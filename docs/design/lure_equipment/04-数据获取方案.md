## 🎯 数据获取策略

### 渐进式数据收集

```
阶段0（Day 1-2）：手动录入10-20款核心产品
    ↓
阶段1（Day 3-5）：爬虫补充50款常见产品
    ↓
阶段2（Day 6-8）：API集成扩展到100+款
    ↓
运营期：基于用户查询按需补充
```

**理由**：
- ✅ 快速验证功能（10款即可开始测试）
- ✅ 降低初期投入（避免大量数据录入）
- ✅ 数据驱动（根据用户实际查询补充）

## 📦 数据源设计

### 三层数据源架构

```python
# src/tools/lure/data_fetcher.py

class LureDataFetcher:
    """统一数据获取接口"""

    def __init__(self, source_type: str = "auto"):
        """
        Args:
            source_type: 数据源类型
                - "auto": 自动降级（API → 爬虫 → 本地）
                - "api": 只用API
                - "scraper": 只用爬虫
                - "local": 只用本地数据库
        """
        self.source_type = source_type
        self.db = ProductionSQLiteDB()

        # 初始化数据源
        if source_type in ["auto", "api"]:
            self.api_client = LureAPIClient()

        if source_type in ["auto", "scraper"]:
            self.scraper = LureWebScraper()

    def fetch_equipment(
        self,
        equipment_type: str,
        filters: Dict
    ) -> List[Dict]:
        """
        统一数据获取接口

        Args:
            equipment_type: 装备类型
            filters: 过滤条件 {"预算": 300, "品牌": "禧玛诺"}

        Returns:
            装备列表
        """
        # 1. 先查本地数据库
        local_results = self._fetch_from_local(equipment_type, filters)

        # 如果本地结果充足，直接返回
        if len(local_results) >= 5:
            return local_results

        # 2. 本地数据不足，尝试外部数据源
        if self.source_type == "auto":
            external_results = self._fetch_with_fallback(equipment_type, filters)
        elif self.source_type == "api":
            external_results = self._fetch_from_api(equipment_type, filters)
        elif self.source_type == "scraper":
            external_results = self._fetch_from_scraper(equipment_type, filters)
        else:
            external_results = []

        # 3. 合并结果并去重
        all_results = local_results + external_results
        unique_results = self._deduplicate(all_results)

        # 4. 将新数据写入本地数据库（异步）
        self._save_to_local_async(external_results)

        return unique_results

    def _fetch_with_fallback(self, equipment_type: str, filters: Dict) -> List[Dict]:
        """降级策略"""
        try:
            # 尝试API（快速）
            results = self._fetch_from_api(equipment_type, filters)
            if results:
                return results
        except Exception as e:
            logger.warning(f"API获取失败: {e}")

        try:
            # 降级到爬虫（慢但可靠）
            results = self._fetch_from_scraper(equipment_type, filters)
            if results:
                return results
        except Exception as e:
            logger.warning(f"爬虫获取失败: {e}")

        # 全部失败，返回空
        return []
```

```python
# src/tools/lure/scraper.py

import requests
from bs4 import BeautifulSoup
import time
import random
from typing import List, Dict, Optional

class LureWebScraper:
    """路亚装备爬虫"""

    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': self._random_user_agent(),
            'Accept': 'text/html,application/xhtml+xml',
            'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.8',
        })

    def _random_user_agent(self) -> str:
        """随机User-Agent"""
        user_agents = [
            'Mozilla/5.0 (Macintosh; Intel Mac OS X 10_15_7)',
            'Mozilla/5.0 (Windows NT 10.0; Win64; x64)',
            'Mozilla/5.0 (X11; Linux x86_64)',
        ]
        return random.choice(user_agents)

    def scrape_decathlon(self, equipment_type: str) -> List[Dict]:
        """
        爬取迪卡侬路亚装备

        优势：
        - 反爬友好（允许爬取）
        - HTML结构清晰
        - 数据质量高

        Args:
            equipment_type: 装备类型

        Returns:
            装备数据列表
        """
        # URL映射
        category_urls = {
            '鱼竿': 'https://www.decathlon.com.cn/zh/browse/c0-sports/c1-fishing/c3-lure-rods',
            '渔轮': 'https://www.decathlon.com.cn/zh/browse/c0-sports/c1-fishing/c3-lure-reels',
            '鱼线': 'https://www.decathlon.com.cn/zh/browse/c0-sports/c1-fishing/c3-fishing-lines',
        }

        url = category_urls.get(equipment_type)
        if not url:
            return []

        try:
            # 请求延迟（礼貌爬取）
            time.sleep(random.uniform(1.0, 3.0))

            response = self.session.get(url, timeout=10)
            response.raise_for_status()

            soup = BeautifulSoup(response.text, 'html.parser')

            # 解析产品列表
            products = []
            for item in soup.select('.product-card'):
                try:
                    product = self._parse_decathlon_product(item, equipment_type)
                    if product:
                        products.append(product)
                except Exception as e:
                    logger.warning(f"解析产品失败: {e}")
                    continue

            logger.info(f"迪卡侬爬取成功: {len(products)}款{equipment_type}")
            return products

        except Exception as e:
            logger.error(f"迪卡侬爬取失败: {e}")
            return []

    def _parse_decathlon_product(self, item, equipment_type: str) -> Optional[Dict]:
        """解析迪卡侬产品"""
        try:
            # 提取基础信息
            name = item.select_one('.product-name').text.strip()
            price_text = item.select_one('.price').text.strip()
            price = float(price_text.replace('¥', '').replace(',', ''))

            # 提取规格（从产品描述）
            specs = self._extract_specs_from_description(
                item.select_one('.product-desc').text,
                equipment_type
            )

            # 生成ID
            product_id = self._generate_id(name, '迪卡侬')

            return {
                'id': product_id,
                'name': name,
                'brand': '迪卡侬',
                'category': equipment_type,
                'price_min': price * 0.95,  # 允许5%价格波动
                'price_max': price * 1.05,
                'specs': specs,
                'data_source': 'scraper_decathlon',
                'last_verified': datetime.now().date().isoformat(),
                'confidence': 0.9
            }

        except Exception as e:
            logger.warning(f"解析失败: {e}")
            return None

    def _extract_specs_from_description(self, description: str, equipment_type: str) -> Dict:
        """从描述文本提取规格"""
        specs = {}

        if equipment_type == '鱼竿':
            # 提取长度（如"2.4米"）
            length_match = re.search(r'(\d+\.?\d*)\s*[米m]', description)
            if length_match:
                specs['length'] = float(length_match.group(1))

            # 提取硬度（如"ML调"）
            power_match = re.search(r'(UUL|UL|L|ML|M|MH|H|XH|XXH)', description)
            if power_match:
                specs['power'] = power_match.group(1)

            # 提取调性
            action_match = re.search(r'([SMF]X?)\s*调', description)
            if action_match:
                specs['action'] = action_match.group(1)

        elif equipment_type == '渔轮':
            # 提取型号（如"C2500"）
            model_match = re.search(r'C?(\d{4})', description)
            if model_match:
                specs['model'] = 'C' + model_match.group(1) if 'C' not in model_match.group(0) else model_match.group(0)

            # 提取齿轮比
            ratio_match = re.search(r'(\d+\.?\d*:\d+)', description)
            if ratio_match:
                specs['gear_ratio'] = ratio_match.group(1)

        return specs

    def _generate_id(self, name: str, brand: str) -> str:
        """生成产品ID"""
        # 简化名称作为ID
        simplified = re.sub(r'[^\w\s]', '', name).replace(' ', '_').lower()
        brand_prefix = re.sub(r'[^\w]', '', brand).lower()
        return f"{brand_prefix}_{simplified}"[:50]
```

```python
# src/tools/lure/api_client.py

from ..utils.api_client import APIClient
from ..utils.cache import cache

class LureAPIClient(APIClient):
    """路亚装备API客户端（预留接口）"""

    def __init__(self, api_key: Optional[str] = None):
        self.api_key = api_key or os.getenv('LURE_API_KEY')
        base_url = os.getenv('LURE_API_URL', 'https://api.lure-data.com')
        super().__init__(base_url=base_url, timeout=10)

        if self.api_key:
            self.session.headers.update({
                'Authorization': f'Bearer {self.api_key}'
            })

    def search_equipment(
        self,
        equipment_type: str,
        filters: Dict
    ) -> Optional[List[Dict]]:
        """
        搜索装备（API调用）

        Args:
            equipment_type: 装备类型
            filters: 过滤条件

        Returns:
            装备列表或None（API失败时）
        """
        # 构建查询参数
        params = {
            'category': equipment_type,
            **self._build_filter_params(filters)
        }

        # 检查缓存
        cache_key = f"lure_api:{equipment_type}:{str(params)}"
        cached_result = cache.get(cache_key)
        if cached_result:
            return cached_result

        # 调用API
        try:
            endpoint = '/v1/equipment/search'
            data = self.get(endpoint, params, use_cache=False)

            if data and data.get('status') == 'success':
                results = data.get('results', [])

                # 转换为标准格式
                standardized = [self._standardize_api_result(r) for r in results]

                # 缓存结果（1小时）
                cache.set(cache_key, standardized, ttl=3600)

                return standardized

            else:
                logger.warning(f"API返回错误: {data}")
                return None

        except Exception as e:
            logger.error(f"API调用失败: {e}")
            return None

    def _build_filter_params(self, filters: Dict) -> Dict:
        """构建API查询参数"""
        params = {}

        if '预算' in filters:
            params['price_max'] = filters['预算']

        if '品牌' in filters:
            params['brand'] = filters['品牌']

        if '硬度' in filters:
            params['power'] = filters['硬度']

        return params

    def _standardize_api_result(self, api_data: Dict) -> Dict:
        """将API数据转换为标准格式"""
        return {
            'id': api_data.get('id'),
            'name': api_data.get('name'),
            'brand': api_data.get('brand'),
            'category': api_data.get('category'),
            'price_min': api_data.get('price', {}).get('min'),
            'price_max': api_data.get('price', {}).get('max'),
            'specs': api_data.get('specifications', {}),
            'data_source': 'api',
            'last_verified': datetime.now().date().isoformat(),
            'confidence': 1.0
        }
```

## ✅ 数据质量控制

### 数据验证

```python
# src/tools/lure/data_validator.py

class DataValidator:
    """数据质量验证"""

    @staticmethod
    def validate_equipment(equipment: Dict) -> Tuple[bool, List[str]]:
        """
        验证装备数据完整性和合理性

        Returns:
            (是否有效, 错误列表)
        """
        errors = []

        # 1. 必填字段检查
        required_fields = ['id', 'name', 'brand', 'category', 'price_min', 'price_max']
        for field in required_fields:
            if field not in equipment or not equipment[field]:
                errors.append(f"缺少必填字段: {field}")

        # 2. 价格合理性检查
        if 'price_min' in equipment and 'price_max' in equipment:
            price_min = equipment['price_min']
            price_max = equipment['price_max']

            if price_min <= 0 or price_max <= 0:
                errors.append(f"价格必须为正数: {price_min}-{price_max}")

            if price_min > price_max:
                errors.append(f"最低价大于最高价: {price_min} > {price_max}")

            if price_max > 50000:
                errors.append(f"价格异常过高: {price_max}")

        # 3. 类别有效性检查
        valid_categories = ['鱼竿', '渔轮', '鱼线', '鱼饵', '鱼钩', '配件', '工具']
        if equipment.get('category') not in valid_categories:
            errors.append(f"无效类别: {equipment.get('category')}")

        # 4. 规格合理性检查
        if equipment.get('category') == '鱼竿':
            specs = equipment.get('specs', {})

            # 长度检查（0.5-5.0米）
            if 'length' in specs:
                length = specs['length']
                if not (0.5 <= length <= 5.0):
                    errors.append(f"鱼竿长度异常: {length}米")

            # 硬度检查
            if 'power' in specs:
                valid_powers = ['UUL', 'UL', 'L', 'ML', 'M', 'MH', 'H', 'XH', 'XXH']
                if specs['power'] not in valid_powers:
                    errors.append(f"无效硬度: {specs['power']}")

        # 5. 数据新鲜度检查
        if 'last_verified' in equipment:
            try:
                last_verified = datetime.fromisoformat(equipment['last_verified'])
                days_old = (datetime.now() - last_verified).days

                if days_old > 180:
                    errors.append(f"数据过期: {days_old}天前")
            except:
                errors.append("无效的验证日期")

        return len(errors) == 0, errors

    @staticmethod
    def clean_data(equipment: Dict) -> Dict:
        """数据清洗"""
        cleaned = equipment.copy()

        # 1. 去除多余空格
        if 'name' in cleaned:
            cleaned['name'] = ' '.join(cleaned['name'].split())

        if 'brand' in cleaned:
            cleaned['brand'] = cleaned['brand'].strip()

        # 2. 价格标准化（保留2位小数）
        if 'price_min' in cleaned:
            cleaned['price_min'] = round(cleaned['price_min'], 2)

        if 'price_max' in cleaned:
            cleaned['price_max'] = round(cleaned['price_max'], 2)

        # 3. 规格标准化
        if 'specs' in cleaned and isinstance(cleaned['specs'], dict):
            # 硬度统一大写
            if 'power' in cleaned['specs']:
                cleaned['specs']['power'] = cleaned['specs']['power'].upper()

            # 长度保留2位小数
            if 'length' in cleaned['specs']:
                cleaned['specs']['length'] = round(cleaned['specs']['length'], 2)

        return cleaned
```

## 📝 数据导入流程

### 批量导入脚本

```python
# scripts/import_equipment_data.py

import json
from src.tools.lure.database import ProductionSQLiteDB
from src.tools.lure.data_validator import DataValidator

def import_from_json(json_file: str):
    """从JSON文件导入装备数据"""
    db = ProductionSQLiteDB()
    validator = DataValidator()

    with open(json_file, 'r', encoding='utf-8') as f:
        data = json.load(f)

    equipments = data.get('equipments', [])

    success_count = 0
    error_count = 0

    for equipment in equipments:
        # 验证数据
        is_valid, errors = validator.validate_equipment(equipment)

        if not is_valid:
            print(f"❌ 验证失败: {equipment.get('name')}")
            for error in errors:
                print(f"   - {error}")
            error_count += 1
            continue

        # 清洗数据
        cleaned = validator.clean_data(equipment)

        # 导入数据库
        try:
            db.insert_equipment(cleaned)
            print(f"✅ 导入成功: {cleaned.get('name')}")
            success_count += 1
        except Exception as e:
            print(f"❌ 导入失败: {cleaned.get('name')} - {e}")
            error_count += 1

    print(f"\n导入完成: 成功{success_count}条, 失败{error_count}条")

if __name__ == '__main__':
    import_from_json('data/initial_equipment.json')
```

## 🔄 数据更新策略

### 定期数据刷新

```python
# scripts/refresh_equipment_data.py

def refresh_outdated_data(days: int = 90):
    """刷新过期数据"""
    db = ProductionSQLiteDB()
    fetcher = LureDataFetcher(source_type="scraper")

    # 查询过期数据
    outdated = db.execute(f'''
        SELECT * FROM equipment
        WHERE julianday('now') - julianday(last_verified) > {days}
    ''')

    print(f"发现 {len(outdated)} 条过期数据")

    for equipment in outdated:
        try:
            # 重新抓取数据
            updated = fetcher.fetch_equipment(
                equipment['category'],
                {'品牌': equipment['brand'], 'id': equipment['id']}
            )

            if updated:
                # 更新数据库
                db.update_equipment(equipment['id'], updated[0])
                print(f"✅ 更新: {equipment['name']}")
            else:
                # 降低置信度（无法更新）
                db.execute_write(
                    "UPDATE equipment SET confidence = confidence * 0.8 WHERE id = ?",
                    (equipment['id'],)
                )
                print(f"⚠️ 无法更新，降低置信度: {equipment['name']}")

        except Exception as e:
            print(f"❌ 更新失败: {equipment['name']} - {e}")
```
