## 1. 设计概述

### 1.1 设计目标

本数据库设计旨在支持以下核心功能：

1. **钓鱼推荐系统**：基于场景、鱼种、预算的智能推荐算法
2. **知识库选购指南**：参数对比、材质分析、选购建议
3. **商业化功能**：价格筛选、购买链接、产品图片展示
4. **专业级参数管理**：66个核心字段覆盖所有关键参数

### 1.2 数据策略

- **关联表策略**：核心筛选参数用独立字段（重量、刹车力、价格）
- **JSON 策略**：复杂非结构化数据用 JSON（适用鱼种、购买链接、产品图片）
- **混合方案**：容线量、齿速比等多值参数使用关联表

### 1.3 核心特性

| 特性 | 描述 |
|------|------|
| **66个专业字段** | 覆盖从基础到发烧友的所有关注点 |
| **三线型容线量** | 尼龙线、PE线、碳线独立管理 |
| **线杯系统详解** | 轴长类型（长轴/短轴）、主轴材质完整记录 ⭐NEW |
| **齿轮系统详解** | 大齿/小齿材质、尺寸、齿数完整记录 |
| **水域适用性** | 淡水/海水/淡海两用，防腐等级评分 |
| **产品图片管理** | 支持多场景图片（主图、细节、场景、参数图） |
| **生命周期管理** | 发布时间、产品代数、生命周期阶段 |

## 2. 数据库表结构

### 2.1 主表 reel_specs

**渔轮规格主表**，包含 66 个核心字段。

#### 完整 SQL 定义

```sql
CREATE TABLE reel_specs (
    -- ========== 基础信息 (5) ==========
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
    brand VARCHAR(100) NOT NULL COMMENT '品牌（Shimano/Daiwa/Abu Garcia等）',
    model VARCHAR(100) NOT NULL COMMENT '型号（Antares DC/Alphas Air TW等）',
    model_size VARCHAR(50) COMMENT '型号尺寸（C2500/C3000/HG/XG等）',
    reel_type ENUM('spinning', 'baitcasting', 'round') NOT NULL COMMENT '渔轮类型：纺车轮/水滴轮/鼓轮',

    -- ========== 水域适用性 (3) ==========
    water_type_compatibility ENUM('freshwater', 'saltwater', 'both') NOT NULL DEFAULT 'freshwater'
        COMMENT '水域适用性：淡水/海水/淡海两用',
    saltwater_rating INT COMMENT '海水性能评分（1-10，10=深海级）',
    corrosion_resistance_level ENUM('basic', 'moderate', 'high', 'marine_grade')
        COMMENT '防腐等级：基础/中等/高级/船钓级',

    -- ========== 尺寸参数 (5) ==========
    weight_g INT COMMENT '自重（克）',
    body_height_mm DECIMAL(5,1) COMMENT '机身高度（毫米）- 水滴轮关键参数',
    body_length_mm DECIMAL(5,1) COMMENT '机身长度（毫米）',
    body_width_mm DECIMAL(5,1) COMMENT '机身宽度（毫米）',
    rotor_diameter_mm INT COMMENT '转子直径（毫米）- 纺车轮用',

    -- ========== 性能参数 (3) ==========
    max_drag_kg DECIMAL(4,1) COMMENT '最大刹车力（公斤）',
    bearings_total INT COMMENT '总轴承数',
    bearings_one_way INT COMMENT '单向轴承数',

    -- ========== 线杯规格 (6) ==========
    spool_diameter_mm INT COMMENT '线杯外径（毫米）',
    spool_width_mm INT COMMENT '线杯宽度（毫米）',
    spool_type ENUM('shallow', 'deep', 'interchangeable') COMMENT '线杯类型：浅线杯/深线杯/可换线杯',
    spool_material VARCHAR(50) COMMENT '线杯材质（铝合金/碳纤维/工程塑料）',
    spool_shaft_type ENUM('long', 'short') COMMENT '线杯轴长类型：长轴(远投)/短轴(精准)',
    spool_shaft_material VARCHAR(50) COMMENT '线杯主轴材质（不锈钢/钛合金/铝合金/碳钢）',

    -- ========== 手柄和摇臂系统 (6) ==========
    handle_type ENUM('single', 'double', 'foldable') COMMENT '手柄类型：单手柄/双手柄/可折叠',
    handle_length_mm INT COMMENT '摇臂总长（毫米）',
    handle_arm_distance_mm INT COMMENT '摇臂轴距（毫米）- 中心轴到握丸中心',
    handle_spacing_mm INT COMMENT '摇柄间距（毫米）- 双手柄用',
    handle_arm_type ENUM('short','standard','long','power') COMMENT '摇臂类型：短柄/标准/长柄/力量型',
    handle_interchangeable BOOLEAN DEFAULT FALSE COMMENT '是否可换手',

    -- ========== 握丸参数 (4) ==========
    handle_knob_material VARCHAR(50) COMMENT '握丸材质（EVA/Cork/TPE/Aluminum/Carbon）',
    handle_knob_size ENUM('S','M','L','XL') COMMENT '握丸尺寸',
    handle_knob_shape ENUM('round','oval','flat','T-shape','custom') COMMENT '握丸形状',
    handle_knob_replaceable BOOLEAN DEFAULT FALSE COMMENT '握丸是否可更换',

    -- ========== 齿轮系统 (8) ==========
    -- 大齿轮（主齿轮/被动齿轮）
    main_gear_material VARCHAR(50) COMMENT '大齿材质（Brass/Stainless/Zinc/Aluminum/Resin）',
    main_gear_diameter_mm DECIMAL(5,2) COMMENT '大齿直径（毫米）',
    main_gear_teeth INT COMMENT '大齿齿数',

    -- 小齿轮（主动齿轮）
    pinion_gear_material VARCHAR(50) COMMENT '小齿材质（Stainless/Brass/Zinc/Carbon Steel）',
    pinion_gear_diameter_mm DECIMAL(5,2) COMMENT '小齿直径（毫米）',
    pinion_gear_teeth INT COMMENT '小齿齿数',

    -- 齿轮系统评价
    gear_durability_rating INT COMMENT '齿轮耐用度评分（1-10）',
    gear_smoothness_rating INT COMMENT '齿轮顺滑度评分（1-10）',

    -- ========== 刹车系统 (2) ==========
    drag_type VARCHAR(50) COMMENT '刹车类型（magnetic/centrifugal/DC/hybrid）',
    drag_position ENUM('front', 'rear', 'dual') COMMENT '刹车位置：前置/后置/双刹',

    -- ========== 材质和工艺 (3) ==========
    body_material VARCHAR(100) COMMENT '车身材质（铝合金/碳纤维/工程塑料/CI4+等）',
    waterproof_rating VARCHAR(20) COMMENT '防水等级（IPX4/IPX6/IPX7/IPX8）',
    anti_reverse VARCHAR(50) COMMENT '逆止系统（infinite/standard/roller bearing）',

    -- ========== 应用场景 JSON (4) ==========
    target_species JSON COMMENT '适用鱼种数组 ["黑鱼","鲈鱼","鳜鱼"]',
    application_scenarios JSON COMMENT '应用场景 {"water":"freshwater","location":"shore,boat"}',
    suitable_techniques JSON COMMENT '适用钓法数组 ["Texas","Jig","Minnow"]',
    bearing_details JSON COMMENT '轴承详情 {"special_types":["9xSCR","2xHSB"],"seal":"MagSealed"}',

    -- ========== 商业信息 (3) ==========
    price_min DECIMAL(10,2) COMMENT '价格区间最低价（元）',
    price_max DECIMAL(10,2) COMMENT '价格区间最高价（元）',
    purchase_links JSON COMMENT '购买链接 [{"platform":"京东","url":"https://..."}]',

    -- ========== 产品生命周期 (6) ==========
    release_year YEAR COMMENT '发布年份',
    release_month TINYINT COMMENT '发布月份（1-12）',
    release_date DATE COMMENT '精确发布日期',
    discontinued_date DATE COMMENT '停产日期',
    generation VARCHAR(50) COMMENT '产品代数（如"第3代"、"2024款"）',
    lifecycle_stage ENUM('new','mature','declining','discontinued') COMMENT '生命周期阶段',

    -- ========== 产品图片 (2) ==========
    images JSON COMMENT '产品图片数组 [{"url":"...","type":"main/detail/scene/spec","description":"..."}]',
    primary_image_url VARCHAR(500) COMMENT '主图URL（冗余字段，用于列表快速展示）',

    -- ========== 附加信息 (3) ==========
    description TEXT COMMENT '产品描述',
    features JSON COMMENT '特色功能数组 ["TWS导线系统","4x8 DC刹车"]',
    overall_dimensions JSON COMMENT '完整尺寸JSON {"height":38,"length":84,"width":51}',

    -- ========== 元数据 (3) ==========
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    is_active BOOLEAN DEFAULT TRUE COMMENT '是否在售',

    -- ========== 索引优化 ==========
    INDEX idx_reel_type (reel_type),
    INDEX idx_brand_model (brand, model),
    INDEX idx_price (price_min, price_max),
    INDEX idx_weight_height (weight_g, body_height_mm),
    INDEX idx_gear_quality (main_gear_material, pinion_gear_material),
    INDEX idx_water_type (water_type_compatibility, saltwater_rating),
    INDEX idx_release_year (release_year),
    INDEX idx_has_images (primary_image_url),
    UNIQUE KEY uk_brand_model_size (brand, model, model_size)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='渔轮规格主表 - 专业版';
```

### 2.2 关联表 reel_line_capacity

**容线量表**，支持三种线型（尼龙线、PE线、碳线）的独立管理。

#### SQL 定义

```sql
CREATE TABLE reel_line_capacity (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
    reel_id BIGINT NOT NULL COMMENT '渔轮ID',

    -- ========== 线型分类（三大类）==========
    line_type ENUM('nylon', 'pe', 'fluorocarbon') NOT NULL
        COMMENT '线型：尼龙线(Nylon)/PE线(Polyethylene)/碳线(Fluorocarbon)',

    -- ========== 强度标注（不同线型用不同字段）==========
    strength_lbs DECIMAL(5,1)
        COMMENT '拉力磅数（LBS）- 尼龙线、碳线常用',
    strength_number DECIMAL(3,1)
        COMMENT '号数（#）- PE线常用、碳线也用',

    -- ========== 容线量 ==========
    length_m INT NOT NULL COMMENT '长度（米）',
    length_yard INT COMMENT '长度（码）',

    -- ========== 线径（可选）==========
    diameter_mm DECIMAL(4,2)
        COMMENT '线径（毫米）- 影响容线量的关键参数',

    -- ========== 附加信息 ==========
    is_recommended BOOLEAN DEFAULT FALSE
        COMMENT '是否厂家推荐规格',
    application_note VARCHAR(200)
        COMMENT '适用说明（如"远投专用"、"打黑推荐"）',

    -- ========== 约束和索引 ==========
    CONSTRAINT chk_strength CHECK (
        (line_type = 'nylon' AND strength_lbs IS NOT NULL) OR
        (line_type = 'pe' AND strength_number IS NOT NULL) OR
        (line_type = 'fluorocarbon' AND (strength_lbs IS NOT NULL OR strength_number IS NOT NULL))
    ),
    FOREIGN KEY (reel_id) REFERENCES reel_specs(id) ON DELETE CASCADE,
    INDEX idx_reel_line (reel_id, line_type),
    INDEX idx_line_type (line_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='渔轮容线量表（三线型）';
```

#### 三线型对比表

| 参数 | 尼龙线（Nylon） | PE线（Polyethylene） | 碳线（Fluorocarbon） |
|------|----------------|---------------------|-------------------|
| **标注方式** | 磅数（LBS） | 号数（#） | 磅数或号数 |
| **线径** | 粗（相同强度下） | 细（同强度直径最小） | 中等偏粗 |
| **容线量** | 中等 | 最大（直径细） | 最小（密度大） |
| **比重** | 1.14（慢沉） | 0.97（浮水） | 1.78（快沉） |
| **延展性** | 高（20-30%） | 几乎无（<3%） | 低（10-15%） |
| **耐磨性** | 中等 | 差（怕硬物） | 优秀 |
| **可见度** | 高 | 中等 | 低（水中隐形） |
| **价格** | 低 | 中高 | 高 |
| **主要用途** | 入门、远投、表层 | 精细路亚、打黑 | 前导线、清水 |
| **常用规格** | 8-16 LBS | 0.6-3.0 # | 6-20 LBS |

### 2.3 关联表 reel_gear_ratios

**齿速比表**，支持单速/双速/三速渔轮的多档位管理。

#### SQL 定义

```sql
CREATE TABLE reel_gear_ratios (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
    reel_id BIGINT NOT NULL COMMENT '渔轮ID',
    gear_name VARCHAR(50) COMMENT '档位名称（high/low/medium）',
    gear_ratio DECIMAL(4,2) NOT NULL COMMENT '齿速比（如7.3:1存为7.3）',
    retrieve_per_turn_cm INT COMMENT '每转卷线长度（厘米）',
    is_default BOOLEAN DEFAULT FALSE COMMENT '是否默认档位',

    FOREIGN KEY (reel_id) REFERENCES reel_specs(id) ON DELETE CASCADE,
    INDEX idx_reel_gear (reel_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='渔轮齿速比表（支持多档位）';
```

## 3. 核心参数详解

### 3.1 机身高度（body_height_mm）⭐⭐⭐⭐⭐

**为什么重要**：影响握持舒适度和长时间作钓疲劳度。

#### 水滴轮高度分类

| 高度范围 | 类型 | 特点 | 适用场景 |
|---------|------|------|---------|
| **30-35mm** | 超低矮型 | 单手握持极佳，抛投灵敏 | 精细路亚、马口/白条 |
| **36-40mm** | 标准型 | 平衡性最佳，通用性强 | 鲈鱼/鳜鱼/翘嘴（主流） |
| **41-45mm** | 高身型 | 容线量大，拖力强 | 打黑/雷强、大型鲈鱼 |

#### 纺车轮体积分类

| 型号 | 机身高度 | 转子直径 | 适用场景 |
|------|---------|---------|---------|
| C2000 | 50-55mm | 40-50mm | 轻量路亚、马口 |
| C3000 | 58-65mm | 55-65mm | 常规路亚、鲈鱼 |
| C5000 | 70-80mm | 75-85mm | 大型路亚、海水 |

---

### 3.2 握丸材质（handle_knob_material）⭐⭐⭐⭐

**为什么重要**：直接接触手掌，影响握持手感、防滑性和疲劳度。

#### 材质对比表

| 材质 | 手感 | 防滑性 | 耐用度 | 重量 | 价位 | 适用场景 |
|------|------|--------|--------|------|------|---------|
| **EVA泡棉** | ⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | ⭐⭐⭐ | 很轻 | 低 | 淡水路亚、长时间作钓 |
| **软木（Cork）** | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐ | ⭐⭐⭐⭐ | 轻 | 高 | 高端纺车轮、飞蝇钓 |
| **TPE橡胶** | ⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 中等 | 中 | 海水路亚、雷强 |
| **铝合金** | ⭐⭐⭐ | ⭐⭐ | ⭐⭐⭐⭐⭐ | 重 | 高 | 精密水滴轮、收藏级 |
| **碳纤维** | ⭐⭐⭐⭐ | ⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 很轻 | 很高 | 旗舰级产品 |
| **ABS塑料** | ⭐⭐ | ⭐⭐ | ⭐⭐⭐ | 轻 | 低 | 入门级产品 |

---

### 3.3 齿轮系统（gear_system）⭐⭐⭐⭐⭐

**为什么重要**：齿轮是渔轮的"心脏"，决定耐用度、顺滑度和维护成本。

#### 大齿材质对比

| 材质 | 耐用度 | 顺滑度 | 重量 | 价位 | 适用场景 |
|------|--------|--------|------|------|---------|
| **黄铜（Brass）** | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 重 | 高端 | 旗舰级，搏大鱼 |
| **不锈钢（Stainless）** | ⭐⭐⭐⭐ | ⭐⭐⭐⭐ | 中等 | 中高端 | 海水防锈，通用 |
| **锌合金（Zinc Alloy）** | ⭐⭐⭐ | ⭐⭐⭐ | 中等 | 中端 | 淡水路亚，性价比 |
| **铝合金（Aluminum）** | ⭐⭐ | ⭐⭐⭐ | 轻 | 入门 | 轻量化设计 |
| **工程塑料（Resin）** | ⭐ | ⭐⭐ | 很轻 | 入门 | 入门级产品 |

#### 齿轮尺寸影响

- **大齿直径**：
  - 水滴轮：25-35mm（越大扭矩越大、耐用度越高）
  - 纺车轮：30-50mm（随型号增大）

- **齿速比计算**：
  ```
  齿速比 = 大齿齿数 ÷ 小齿齿数
  例如：84齿 ÷ 12齿 = 7.0:1
  ```

---

### 3.4 线杯系统（spool_system）⭐⭐⭐⭐

**为什么重要**：线杯是渔轮的核心部件，轴长和主轴材质直接影响抛投性能和耐用度。

#### 线杯轴长类型（spool_shaft_type）

| 类型 | 特点 | 优势 | 劣势 | 适用场景 |
|------|------|------|------|---------|
| **长轴（Long Cast）** | 线杯较深，轴心距离远 | 远投性能好，容线量大 | 精准度稍差，重量大 | 海钓、远投、大型鱼种 |
| **短轴（Short Cast）** | 线杯较浅，轴心距离近 | 抛投精准，操控灵敏 | 容线量小，远投受限 | 精细路亚、准确抛投、淡水 |

**选择建议**：
- **纺车轮长轴（LC）**：适合岸抛、远投、海水路亚（C3000LC、C5000XG-LC）
- **纺车轮短轴（SC）**：适合溪流、小水面、精细作钓（C2000SC、C2500SC）
- **水滴轮**：通常不区分长短轴，但线杯深浅会影响类似特性

#### 线杯主轴材质（spool_shaft_material）

| 材质 | 强度 | 耐腐蚀 | 重量 | 价位 | 适用场景 |
|------|------|--------|------|------|---------|
| **不锈钢（Stainless Steel）** | ⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 中等 | 中端 | 通用、性价比最佳、淡海两用 |
| **钛合金（Titanium）** | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 很轻 | 高端 | 旗舰级、轻量化、海水 |
| **铝合金（Aluminum）** | ⭐⭐⭐ | ⭐⭐ | 轻 | 入门 | 淡水、入门级 |
| **碳钢（Carbon Steel）** | ⭐⭐⭐⭐⭐ | ⭐ | 重 | 中端 | 淡水、高强度需求 |

**材质影响**：
- **耐用度**：钛合金 > 不锈钢 > 碳钢 > 铝合金
- **防腐蚀**：钛合金 = 不锈钢 >>> 碳钢 > 铝合金
- **轻量化**：钛合金 > 铝合金 > 不锈钢 > 碳钢

## 6. 推荐算法示例

### 6.1 舒适度评分算法（水滴轮）

```python
def calculate_comfort_score(reel):
    """
    计算水滴轮的舒适度评分（0-100分）

    权重分配：
    - 机身高度：40%（最关键）
    - 重量：35%
    - 摇臂轴距：15%
    - 握丸材质：10%
    """

    # 高度评分（理想值：33-36mm）
    height = reel.body_height_mm
    if height <= 33:
        height_score = 100
    elif height <= 36:
        height_score = 100 - (height - 33) * 5
    else:
        height_score = max(0, 85 - (height - 36) * 3)

    # 重量评分（理想值：≤150g）
    weight = reel.weight_g
    weight_score = max(0, 100 - (weight - 120) * 0.5)

    # 摇臂轴距评分（理想值：85-90mm）
    arm_distance = reel.handle_arm_distance_mm
    if 85 <= arm_distance <= 90:
        arm_score = 100
    else:
        arm_score = max(0, 100 - abs(arm_distance - 87.5) * 2)

    # 握丸材质评分
    knob_material_scores = {
        'EVA': 95,
        'Cork': 100,
        'TPE': 90,
        'Aluminum': 70,
        'Carbon': 85,
        'ABS': 60
    }
    knob_score = knob_material_scores.get(reel.handle_knob_material, 70)

    # 综合评分
    total_score = (
        height_score * 0.40 +
        weight_score * 0.35 +
        arm_score * 0.15 +
        knob_score * 0.10
    )

    return round(total_score, 2)

# 示例
# Daiwa Alphas Air TW:
# - height: 35.5mm → 92.5分
# - weight: 128g → 96分
# - arm: 88mm → 99分
# - knob: EVA → 95分
# = 92.5*0.4 + 96*0.35 + 99*0.15 + 95*0.1 = 94.95分（极高舒适度）
```

### 6.2 海水适用性评分算法

```python
def calculate_saltwater_suitability_score(reel):
    """
    计算海水适用性评分（0-100分）

    权重分配：
    - 水域兼容性：40%
    - 海水评分：25%
    - 防腐等级：20%
    - 防水等级：15%
    """

    score = 0

    # 水域兼容性（40%）
    if reel.water_type == 'saltwater':
        score += 40
    elif reel.water_type == 'both':
        score += 35
    else:
        return 0  # 淡水轮不推荐海钓

    # 海水评分（25%）
    if reel.saltwater_rating:
        score += reel.saltwater_rating * 2.5

    # 防腐等级（20%）
    corrosion_scores = {
        'marine_grade': 20,
        'high': 15,
        'moderate': 8,
        'basic': 3
    }
    score += corrosion_scores.get(reel.corrosion_resistance_level, 0)

    # 防水等级（15%）
    waterproof_scores = {
        'IPX8': 15,
        'IPX7': 12,
        'IPX6': 8,
        'IPX4': 4
    }
    score += waterproof_scores.get(reel.waterproof_rating, 0)

    return score

# 示例
# Daiwa Saltiga:
# - water_type: saltwater → 40分
# - saltwater_rating: 10 → 25分
# - corrosion: marine_grade → 20分
# - waterproof: IPX8 → 15分
# = 100分（完美海水轮）
```

### 6.3 耐用度评分算法

```python
def calculate_durability_score(reel):
    """
    计算渔轮耐用度评分（0-100分）

    权重分配：
    - 大齿材质：35%
    - 小齿材质：20%
    - 轴承数量：15%
    - 齿轮耐用度评分：20%
    - 车身材质：10%
    """

    # 大齿材质评分
    main_gear_scores = {
        'Brass': 100,
        'Stainless Steel': 85,
        'Zinc Alloy': 70,
        'Aluminum': 50,
        'Resin': 30
    }
    main_gear_score = main_gear_scores.get(reel.main_gear_material, 60)

    # 小齿材质评分
    pinion_gear_scores = {
        'Stainless Steel': 100,
        'Brass': 90,
        'Carbon Steel': 85,
        'Zinc Alloy': 65
    }
    pinion_gear_score = pinion_gear_scores.get(reel.pinion_gear_material, 70)

    # 轴承数量评分（10-15个为最佳）
    bearings = reel.bearings_total
    if 10 <= bearings <= 15:
        bearings_score = 100
    elif bearings < 10:
        bearings_score = max(0, 50 + bearings * 5)
    else:
        bearings_score = max(70, 100 - (bearings - 15) * 3)

    # 齿轮耐用度评分（厂家提供）
    gear_rating_score = (reel.gear_durability_rating or 7) * 10

    # 车身材质评分
    body_material_scores = {
        'CI4+ Carbon': 100,
        'Carbon Composite': 95,
        'Aluminum Alloy': 90,
        'Magnesium Alloy': 85,
        'ABS': 60
    }
    # 简单匹配（实际应更智能）
    body_score = 80  # 默认值
    for material, score in body_material_scores.items():
        if material in reel.body_material:
            body_score = score
            break

    # 综合评分
    total_score = (
        main_gear_score * 0.35 +
        pinion_gear_score * 0.20 +
        bearings_score * 0.15 +
        gear_rating_score * 0.20 +
        body_score * 0.10
    )

    return round(total_score, 2)

# 示例
# Shimano Antares DC MD:
# - main_gear: Brass → 100分
# - pinion_gear: Stainless Steel → 100分
# - bearings: 10 → 100分
# - gear_rating: 10 → 100分
# - body: CI4+ Carbon → 100分
# = 100分（极高耐用度）
```
