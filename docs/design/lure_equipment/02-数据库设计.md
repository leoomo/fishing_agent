## 📊 数据库选型

### SQLite + WAL模式

**选型理由**：
- ✅ 与项目fishing-agent保持一致（已用SQLite）
- ✅ 零配置部署（单文件数据库）
- ✅ 数据规模匹配（100-500款装备，<5MB）
- ✅ WAL模式支持并发读（99%读场景）
- ✅ 性能足够（支持50+ QPS）

**配置要求**：
```sql
PRAGMA journal_mode=WAL;           -- 核心：写前日志，支持并发读
PRAGMA synchronous=NORMAL;         -- 性能优化
PRAGMA cache_size=-64000;          -- 64MB缓存
PRAGMA busy_timeout=30000;         -- 锁等待30秒
PRAGMA temp_store=MEMORY;          -- 临时表放内存
```

## 🗄️ 表结构设计

### 1. equipment（装备主表）

存储所有装备的基础信息。

```sql
CREATE TABLE equipment (
    id TEXT PRIMARY KEY,              -- 装备唯一ID（如：shimano_poison_264mh）
    name TEXT NOT NULL,                -- 产品名称
    brand TEXT NOT NULL,               -- 品牌名称
    category TEXT NOT NULL,            -- 装备类别（鱼竿/渔轮/鱼线/鱼饵/鱼钩/配件/工具）
    price_min REAL,                    -- 价格区间最低值
    price_max REAL,                    -- 价格区间最高值
    specs_json TEXT,                   -- 规格参数（JSON格式，存储可变字段）
    data_source TEXT,                  -- 数据来源（manual/api/scraper）
    last_verified DATE,                -- 最后验证日期
    confidence REAL DEFAULT 1.0,       -- 数据置信度（0.0-1.0）
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 索引优化
CREATE INDEX idx_equipment_category ON equipment(category);
CREATE INDEX idx_equipment_brand ON equipment(brand);
CREATE INDEX idx_equipment_price ON equipment(price_min, price_max);
CREATE INDEX idx_equipment_verified ON equipment(last_verified);
```

**字段说明**：

| 字段 | 类型 | 说明 | 示例 |
|------|------|------|------|
| id | TEXT | 唯一标识符 | `shimano_poison_264mh` |
| name | TEXT | 产品全称 | `禧玛诺毒牙21代 264MH` |
| brand | TEXT | 品牌名称 | `禧玛诺`/`Shimano` |
| category | TEXT | 装备分类 | `鱼竿`/`渔轮`/`鱼线` |
| price_min/max | REAL | 价格区间 | 650.0 / 750.0 |
| specs_json | TEXT | JSON规格 | `{"length": 2.64, "power": "MH"}` |
| data_source | TEXT | 数据来源标记 | `manual`/`api`/`scraper` |
| last_verified | DATE | 数据验证日期 | `2025-11-20` |
| confidence | REAL | 数据质量评分 | 0.0-1.0 (1.0最高) |

### 2. rod_specs（鱼竿规格表）

将鱼竿的高频查询字段独立出来，优化查询性能。

```sql
CREATE TABLE rod_specs (
    equipment_id TEXT PRIMARY KEY,     -- 关联equipment.id
    length REAL,                       -- 长度（米）
    power TEXT,                        -- 硬度（UL/L/ML/M/MH/H/XH）
    action TEXT,                       -- 调性（S/M/MF/F/XF）
    material TEXT,                     -- 材质（如：36T碳素）
    weight REAL,                       -- 自重（克）
    lure_weight_min REAL,              -- 饵重范围最小值（克）
    lure_weight_max REAL,              -- 饵重范围最大值（克）
    FOREIGN KEY (equipment_id) REFERENCES equipment(id) ON DELETE CASCADE
);

-- 索引优化（高频查询字段）
CREATE INDEX idx_rod_power ON rod_specs(power);
CREATE INDEX idx_rod_length ON rod_specs(length);
CREATE INDEX idx_rod_action ON rod_specs(action);
```

**为什么单独建表**？
- 鱼竿的"硬度"、"长度"是最高频查询条件
- 独立表 + 索引 → 查询速度提升10x
- 其他装备类别查询频率低，保持JSON即可

## 🔧 数据库初始化代码

```python
# src/tools/lure/database.py

import sqlite3
from pathlib import Path
from typing import Optional
import threading

class ProductionSQLiteDB:
    """生产级SQLite数据库"""

    _instance = None
    _lock = threading.Lock()

    def __new__(cls, *args, **kwargs):
        """单例模式"""
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self, db_path: str = None):
        if hasattr(self, '_initialized'):
            return

        if db_path is None:
            db_path = Path(__file__).parent / "data" / "equipment.db"

        self.db_path = db_path
        self._local = threading.local()  # 线程本地连接
        self._init_database()
        self._initialized = True

    def _create_connection(self):
        """创建数据库连接"""
        conn = sqlite3.connect(
            self.db_path,
            timeout=30.0,
            check_same_thread=False
        )
        conn.row_factory = sqlite3.Row  # 返回字典格式

        # ⭐ 关键配置 ⭐
        conn.execute("PRAGMA journal_mode=WAL")
        conn.execute("PRAGMA synchronous=NORMAL")
        conn.execute("PRAGMA cache_size=-64000")
        conn.execute("PRAGMA busy_timeout=30000")
        conn.execute("PRAGMA temp_store=MEMORY")
        conn.execute("PRAGMA foreign_keys=ON")

        return conn

    def _get_connection(self):
        """获取线程本地连接"""
        if not hasattr(self._local, 'connection'):
            self._local.connection = self._create_connection()
        return self._local.connection

    def _init_database(self):
        """初始化数据库表结构"""
        conn = self._create_connection()
        cursor = conn.cursor()

        # 创建equipment表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS equipment (
                id TEXT PRIMARY KEY,
                name TEXT NOT NULL,
                brand TEXT NOT NULL,
                category TEXT NOT NULL,
                price_min REAL,
                price_max REAL,
                specs_json TEXT,
                data_source TEXT,
                last_verified DATE,
                confidence REAL DEFAULT 1.0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

        # 创建索引
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_equipment_category ON equipment(category)')
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_equipment_brand ON equipment(brand)')
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_equipment_price ON equipment(price_min, price_max)')

        # 创建rod_specs表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS rod_specs (
                equipment_id TEXT PRIMARY KEY,
                length REAL,
                power TEXT,
                action TEXT,
                material TEXT,
                weight REAL,
                lure_weight_min REAL,
                lure_weight_max REAL,
                FOREIGN KEY (equipment_id) REFERENCES equipment(id) ON DELETE CASCADE
            )
        ''')

        cursor.execute('CREATE INDEX IF NOT EXISTS idx_rod_power ON rod_specs(power)')
        cursor.execute('CREATE INDEX IF NOT EXISTS idx_rod_length ON rod_specs(length)')

        # 创建brands表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS brands (
                brand_name TEXT PRIMARY KEY,
                reputation_score REAL DEFAULT 50.0,
                country TEXT,
                tier TEXT,
                is_premium BOOLEAN DEFAULT 0,
                description TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

        conn.commit()
        conn.close()
```

### 2. 鱼竿专用查询（高频优化）

```python
def search_rods(self, power: Optional[str] = None, length_min: Optional[float] = None):
    """查询鱼竿（优化版）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT e.*, r.length, r.power, r.action, r.material
        FROM equipment e
        JOIN rod_specs r ON e.id = r.equipment_id
        WHERE e.category = '鱼竿'
    '''
    params = []

    if power:
        query += ' AND r.power = ?'
        params.append(power)

    if length_min:
        query += ' AND r.length >= ?'
        params.append(length_min)

    query += ' ORDER BY e.confidence DESC LIMIT 20'

    cursor.execute(query, params)
    return [dict(row) for row in cursor.fetchall()]
```

**性能优化**：
- JOIN查询合并equipment和rod_specs
- 使用 `idx_rod_power` 和 `idx_rod_length` 索引
- 预计查询时间：< 3ms

## 🔒 并发处理方案

### 问题：SQLite文件锁

**默认模式问题**：
```
写操作 → EXCLUSIVE锁 → 阻塞所有读写
读操作 → SHARED锁 → 被写操作阻塞
```

**WAL模式解决**：
```
写操作 → 写入.wal文件 → 不阻塞读操作 ✅
读操作 → 从主库+.wal文件读取 → 不被写阻塞 ✅
```

### 线程安全设计

```python
import threading

class ProductionSQLiteDB:
    def __init__(self):
        self._local = threading.local()  # 每个线程独立连接

    def _get_connection(self):
        """线程本地连接"""
        if not hasattr(self._local, 'connection'):
            self._local.connection = self._create_connection()
        return self._local.connection
```

**好处**：
- 每个线程有独立连接
- 避免连接共享导致的竞态条件
- 支持10+并发线程

### 事务与重试机制

```python
from contextlib import contextmanager
import time

@contextmanager
def transaction(self):
    """事务上下文管理器（自动重试）"""
    conn = self._get_connection()
    max_retries = 3

    for attempt in range(max_retries):
        try:
            conn.execute("BEGIN IMMEDIATE")  # 立即获取写锁
            yield conn
            conn.commit()
            return

        except sqlite3.OperationalError as e:
            conn.rollback()
            if "locked" in str(e) and attempt < max_retries - 1:
                time.sleep(0.1 * (2 ** attempt))  # 指数退避
                continue
            else:
                raise

# 使用示例
with db.transaction() as conn:
    cursor = conn.cursor()
    cursor.execute("INSERT INTO equipment VALUES (...)")
```

### 数据更新策略

```python
def update_equipment(self, equipment_id: str, updates: Dict):
    """更新装备数据"""
    with self.transaction() as conn:
        cursor = conn.cursor()

        set_clause = ', '.join([f"{k} = ?" for k in updates.keys()])
        values = list(updates.values()) + [equipment_id]

        cursor.execute(f'''
            UPDATE equipment
            SET {set_clause}, updated_at = CURRENT_TIMESTAMP
            WHERE id = ?
        ''', values)
```

### 数据清理

```python
def clean_outdated_data(self, days: int = 180):
    """清理过期数据"""
    with self.transaction() as conn:
        cursor = conn.cursor()

        # 标记过期数据（不直接删除）
        cursor.execute('''
            UPDATE equipment
            SET confidence = confidence * 0.5
            WHERE julianday('now') - julianday(last_verified) > ?
        ''', (days,))

        print(f"标记了 {cursor.rowcount} 条过期数据")
```

### 备份与恢复

```python
def backup_database(self, backup_path: str):
    """备份数据库"""
    import shutil

    # WAL模式需要checkpoint
    conn = self._get_connection()
    conn.execute("PRAGMA wal_checkpoint(FULL)")

    # 复制文件
    shutil.copy2(self.db_path, backup_path)
    print(f"数据库已备份到: {backup_path}")
```
