```python
# src/tools/lure/database.py

import sqlite3
from pathlib import Path
from typing import Optional, List, Dict, Any
import threading
import json

class ProductionSQLiteDB:
    """生产级SQLite数据库（v4.0模块化版）"""

    _instance = None
    _lock = threading.Lock()

    def __new__(cls, *args, **kwargs):
        """单例模式"""
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self, db_path: str = None):
        """初始化数据库"""
        if hasattr(self, '_initialized'):
            return

        self.db_path = db_path or Path(__file__).parent / "data" / "equipment.db"
        self._local = threading.local()  # 线程本地连接
        self._init_database()
        self._initialized = True
```

```python
def _create_connection(self) -> sqlite3.Connection:
    """创建数据库连接"""
    conn = sqlite3.connect(
        self.db_path,
        timeout=30.0,
        check_same_thread=False
    )
    conn.row_factory = sqlite3.Row  # 返回字典格式

    # WAL模式配置
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA synchronous=NORMAL")
    conn.execute("PRAGMA cache_size=-64000")
    conn.execute("PRAGMA busy_timeout=30000")
    conn.execute("PRAGMA temp_store=MEMORY")
    conn.execute("PRAGMA foreign_keys=ON")

    return conn
```

### 线程本地连接

```python
def _get_connection(self) -> sqlite3.Connection:
    """获取线程本地连接"""
    if not hasattr(self._local, 'connection'):
        self._local.connection = self._create_connection()
    return self._local.connection
```

```python
def search_accessories(
    self,
    accessory_category: Optional[str] = None,
    accessory_type_id: Optional[str] = None,
    material: Optional[str] = None,
    weight_min: Optional[float] = None,
    weight_max: Optional[float] = None,
    is_eco_friendly: Optional[bool] = None,
    price_max: Optional[float] = None,
    limit: int = 20
) -> List[Dict[str, Any]]:
    """查询配件（详见配件表设计文档）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT e.*, a.*, at.name_cn as type_name, at.category, b.reputation_score
        FROM equipment e
        JOIN accessory_specs a ON e.id = a.equipment_id
        JOIN accessory_types at ON a.accessory_type_id = at.id
        LEFT JOIN brands b ON e.brand = b.brand_name
        WHERE e.category = '配件'
    '''
    params = []

    if accessory_category:
        query += ' AND at.category = ?'
        params.append(accessory_category)

    if accessory_type_id:
        query += ' AND a.accessory_type_id = ?'
        params.append(accessory_type_id)

    if material:
        query += ' AND a.material LIKE ?'
        params.append(f'%{material}%')

    if weight_min is not None:
        query += ' AND a.weight_g >= ?'
        params.append(weight_min)

    if weight_max is not None:
        query += ' AND a.weight_g <= ?'
        params.append(weight_max)

    if is_eco_friendly is not None:
        query += ' AND a.is_eco_friendly = ?'
        params.append(1 if is_eco_friendly else 0)

    if price_max:
        query += ' AND e.price_min <= ?'
        params.append(price_max)

    query += ' ORDER BY b.reputation_score DESC, e.confidence DESC LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)

    results = []
    for row in cursor.fetchall():
        accessory = dict(row)
        if accessory.get('features_2025'):
            accessory['features_2025'] = json.loads(accessory['features_2025'])
        results.append(accessory)

    return results

def search_hooks(
    self,
    hook_type: Optional[str] = None,
    hook_size: Optional[str] = None,
    is_angle_adjustable: Optional[bool] = None,
    is_biodegradable: Optional[bool] = None,
    limit: int = 20
) -> List[Dict[str, Any]]:
    """查询鱼钩（详见配件表设计文档）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT e.*, a.*, h.*, at.name_cn as type_name
        FROM equipment e
        JOIN accessory_specs a ON e.id = a.equipment_id
        JOIN hook_specs h ON e.id = h.equipment_id
        JOIN accessory_types at ON a.accessory_type_id = at.id
        WHERE at.category = '钩类'
    '''
    params = []

    if hook_type:
        query += ' AND h.hook_type = ?'
        params.append(hook_type)

    if hook_size:
        query += ' AND h.hook_size = ?'
        params.append(hook_size)

    if is_angle_adjustable is not None:
        query += ' AND h.is_angle_adjustable = ?'
        params.append(1 if is_angle_adjustable else 0)

    if is_biodegradable is not None:
        query += ' AND h.is_biodegradable = ?'
        params.append(1 if is_biodegradable else 0)

    query += ' ORDER BY e.confidence DESC LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)
    return [dict(row) for row in cursor.fetchall()]

def search_weights_by_range(
    self,

    weight_max: float,

    cursor.execute(query, (weight_min, weight_max, limit))
    return [dict(row) for row in cursor.fetchall()]
```

```python
def search_rigs(
    self,
    rig_category: Optional[str] = None,
    difficulty_level: Optional[str] = None,
    target_species: Optional[str] = None,
    water_type: Optional[str] = None,
    scenario: Optional[str] = None,
    limit: int = 20
) -> List[Dict[str, Any]]:
    """查询钓组（详见钓组表设计文档）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT rt.*, rs.*
        FROM rig_types rt
        LEFT JOIN rig_specs rs ON rt.id = rs.rig_type_id
        WHERE 1=1
    '''
    params = []

    if rig_category:
        query += ' AND rt.category = ?'
        params.append(rig_category)

    if difficulty_level:
        query += ' AND rt.difficulty_level = ?'
        params.append(difficulty_level)

    if target_species:
        query += ' AND rs.target_species LIKE ?'
        params.append(f'%{target_species}%')

    if water_type:
        query += ' AND rs.water_type = ?'
        params.append(water_type)

    if scenario:
        query += ' AND rs.suitable_scenario LIKE ?'
        params.append(f'%{scenario}%')

    query += ' ORDER BY rt.difficulty_level, rt.id LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)

    results = []
    for row in cursor.fetchall():
        rig = dict(row)
        if rig.get('target_species'):
            rig['target_species'] = json.loads(rig['target_species'])
        if rig.get('suitable_scenario'):
            rig['suitable_scenario'] = json.loads(rig['suitable_scenario'])
        if rig.get('trend_2025'):
            rig['trend_2025'] = json.loads(rig['trend_2025'])
        results.append(rig)

    return results

def get_rig_components(
    self,
    rig_type_id: str
) -> Dict[str, Any]:
    """获取钓组的配件组成（详见钓组表设计文档）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    # 获取钓组基本信息
    cursor.execute('''
        SELECT rt.*, rs.*
        FROM rig_types rt
        LEFT JOIN rig_specs rs ON rt.id = rs.rig_type_id
        WHERE rt.id = ?
    ''', (rig_type_id,))

    rig_info = cursor.fetchone()
    if not rig_info:
        return None

    # 获取配件组成
    cursor.execute('''
        SELECT rc.*, at.name_cn, at.category, at.name_en
        FROM rig_components rc
        JOIN accessory_types at ON rc.accessory_type_id = at.id
        WHERE rc.rig_type_id = ?
        ORDER BY rc.is_required DESC, rc.component_role
    ''', (rig_type_id,))

    components = [dict(row) for row in cursor.fetchall()]

    result = dict(rig_info)
    result['components'] = components

    # 解析JSON字段
    if result.get('target_species'):
        result['target_species'] = json.loads(result['target_species'])
    if result.get('suitable_scenario'):
        result['suitable_scenario'] = json.loads(result['suitable_scenario'])
    if result.get('trend_2025'):
        result['trend_2025'] = json.loads(result['trend_2025'])

    return result

def recommend_rigs_for_scenario(
    self,
    target_species: str,
    water_type: str = None,
    difficulty_level: str = '入门',
    limit: int = 5
) -> List[Dict[str, Any]]:
    """根据场景推荐钓组（详见钓组表设计文档）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT rt.*, rs.*,
               CASE
                   WHEN rt.difficulty_level = ? THEN 3
                   WHEN rt.difficulty_level = '进阶' THEN 2
                   ELSE 1
               END as match_score
        FROM rig_types rt
        LEFT JOIN rig_specs rs ON rt.id = rs.rig_type_id
        WHERE rs.target_species LIKE ?
    '''
    params = [difficulty_level, f'%{target_species}%']

    if water_type:
        query += ' AND rs.water_type = ?'
        params.append(water_type)

    query += ' ORDER BY match_score DESC, rt.difficulty_level LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)

    results = []
    for row in cursor.fetchall():
        rig = dict(row)
        if rig.get('target_species'):
            rig['target_species'] = json.loads(rig['target_species'])
        if rig.get('suitable_scenario'):
            rig['suitable_scenario'] = json.loads(rig['suitable_scenario'])
        results.append(rig)

    return results
```

```python
def search_equipment(
    self,
    category: str,
    price_max: Optional[float] = None,
    limit: int = 20
) -> List[Dict[str, Any]]:
    """查询装备（基础查询）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT e.*, b.reputation_score, b.tier
        FROM equipment e
        LEFT JOIN brands b ON e.brand = b.brand_name
        WHERE e.category = ?
    '''
    params = [category]

    if price_max:
        query += ' AND e.price_min <= ?'
        params.append(price_max)

    query += ' ORDER BY b.reputation_score DESC, e.confidence DESC LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)

    results = []
    for row in cursor.fetchall():
        equipment = dict(row)
        if equipment.get('specs_json'):
            equipment['specs'] = json.loads(equipment['specs_json'])
            del equipment['specs_json']
        results.append(equipment)

    return results
```

```python
def add_equipment(self, equipment_data: Dict[str, Any]) -> str:
    """添加装备"""
    conn = self._get_connection()
    cursor = conn.cursor()

    try:
        cursor.execute('''
            INSERT INTO equipment (id, name, brand, category, price_min, price_max,
                                 specs_json, data_source, confidence)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        ''', (
            equipment_data['id'],
            equipment_data['name'],
            equipment_data['brand'],
            equipment_data['category'],
            equipment_data.get('price_min'),
            equipment_data.get('price_max'),
            json.dumps(equipment_data.get('specs', {}), ensure_ascii=False),
            equipment_data.get('data_source', 'manual'),
            equipment_data.get('confidence', 1.0)
        ))
        conn.commit()
        return equipment_data['id']
    except Exception as e:
        conn.rollback()
        raise
```

```python
def update_equipment(self, equipment_id: str, updates: Dict):
    """更新装备数据"""
    with self.transaction() as conn:
        cursor = conn.cursor()

        set_clause = ', '.join([f"{k} = ?" for k in updates.keys()])
        values = list(updates.values()) + [equipment_id]

            UPDATE equipment
            SET {set_clause}, updated_at = CURRENT_TIMESTAMP

        ''', values)
```

```python
        for acc in accessories:
            price = f"{acc['price_min']}-{acc['price_max']}" if acc['price_max'] else str(acc['price_min'])
            print(f"  - {acc['brand']} {acc['name']}: ¥{price}")
        print()
```
