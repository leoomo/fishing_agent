## 1. 设计概述

### 1.1 导环的重要性

导环被誉为路亚竿的"**血管**"和"**咽喉**"，它决定了：
- **钓线传输的顺畅度** - 影响抛投距离和线组寿命
- **竿体的感度传导** - 传递鱼讯信号
- **整竿的平衡配重** - 影响操作手感

> **产品开发关键点**: 导环的选配往往是决定一支竿子**成本和等级（Price Point）**的最关键因素。

### 1.2 表结构概览

导环相关表共5个：

| 表名 | 说明 | 记录数量级 |
|------|------|------------|
| **guide_specs** | 导环规格表（核心表） | ~500+ |
| **guide_brands** | 导环品牌字典表（新增） | ~10 |
| **guide_train_concepts** | 配导系统概念表（新增） | ~5 |
| **rod_guide_kits** | 导环套装表 | ~50+ |
| **rod_guide_kit_items** | 套装组成明细表 | ~300+ |

## 2. guide_specs（导环规格表）

### 2.1 表结构

导环作为独立配件的详细规格表，v5.0 版本根据产品专家知识体系全面升级。

```sql
CREATE TABLE guide_specs (
    equipment_id TEXT PRIMARY KEY,          -- 关联equipment.id

    -- ═══════════════════════════════════════════════════════════
    -- 🏷️ 产品标识
    -- ═══════════════════════════════════════════════════════════
    sku_code TEXT UNIQUE,                   -- SKU编码: 'T-KLTG 20H' (钛架-KL型-Torzite环-20号-高脚)
    brand_id INTEGER,                       -- 关联guide_brands.id
    product_tier TEXT CHECK(product_tier IN ('顶级', '高端', '中端', '入门')),

    -- ═══════════════════════════════════════════════════════════
    -- 🔬 磁环规格 (Ring/Insert) - 核心中的核心
    -- ═══════════════════════════════════════════════════════════
    ring_material TEXT NOT NULL CHECK(ring_material IN (
        'Torzite',          -- 顶级旗舰 (Fuji巅峰之作，比SiC更薄更轻)
        'SiC',              -- 高端标配 (碳化硅，硬度~2200Hv)
        'SiN',              -- 重型海钓 (氮化硅，抗冲击)
        'Alconite',         -- 中端性价比王 (黑环，硬度~1500Hv)
        'FazLite',          -- 入门进阶 (深蓝色环)
        'Zirconia',         -- 非Fuji系主流 (氧化锆)
        'Aluminum_Oxide',   -- 低端入门 (氧化铝/O环)
        '陶瓷',              -- 通用陶瓷
        '不锈钢'             -- 金属环
    )),
    ring_hardness TEXT CHECK(ring_hardness IN ('顶级', '高', '中高', '中', '低')),
    ring_thermal_conductivity TEXT CHECK(ring_thermal_conductivity IN ('极高', '高', '中', '低')),
    ring_inner_diameter REAL CHECK(ring_inner_diameter > 0),    -- 环体内径 (mm)
    ring_outer_diameter REAL CHECK(ring_outer_diameter >= ring_inner_diameter), -- 环体外径 (mm)
    ring_thickness REAL CHECK(ring_thickness > 0),              -- 环体厚度 (mm)

    -- ═══════════════════════════════════════════════════════════
    -- 🔧 支架规格 (Frame)
    -- ═══════════════════════════════════════════════════════════
    frame_material TEXT NOT NULL CHECK(frame_material IN (
        '钛合金',           -- Titanium: 极轻，不生锈，记忆弹性，顶级
        '不锈钢_316',       -- SS316L: 高防锈等级，主流
        '不锈钢_304',       -- SS304: 标准防锈，主流
        '碳纤维',           -- Carbon: 最轻，高刚性 (Daiwa AGS专利)
        '铝合金'            -- Aluminum: 轻量，入门
    )),
    frame_type TEXT NOT NULL CHECK(frame_type IN (
        -- K系列 (行业标准，防缠绕)
        'KW',               -- 双脚K型，强力型，用于枪柄竿/大物竿
        'KL',               -- 单脚K型，高脚
        'KL-H',             -- 单脚K型高脚小口径，KR概念专用
        'KT',               -- 单脚极短，用于竿稍走线环
        'KB',               -- 单脚加宽脚，高负荷腹部位置
        -- 其他系列
        'Y',                -- 传统单脚高支架，旧式/低端
        'MN',               -- 重型双脚包覆式，海钓GT/铁板
        'L',                -- 极简单脚微物，轻量化
        'RV',               -- 钛合金反向双脚，顶级出发环
        -- 通用
        '单脚',
        '双脚',
        '无脚',
        '三脚'
    )),
    frame_coating TEXT,                     -- 涂装: '枪色(Gunsmoke)', '银色', '黑色', '金色'
    frame_height REAL CHECK(frame_height > 0),      -- 支架高度 (mm) - 配环关键数据
    frame_thickness REAL CHECK(frame_thickness > 0), -- 支架厚度 (mm)
    foot_type TEXT CHECK(foot_type IN ('单脚', '双脚', '三脚', '无脚')),
    foot_length REAL CHECK(foot_length > 0),        -- 脚长 (mm)

    -- ═══════════════════════════════════════════════════════════
    -- 📐 尺寸与重量
    -- ═══════════════════════════════════════════════════════════
    guide_type TEXT NOT NULL CHECK(guide_type IN (
        'tip_top',          -- 顶环: 竿稍最前端，必须有防缠绕设计
        'stripper',         -- 出发环/导线环: 离轮座最近的大导环，收束螺旋线
        'choke',            -- 过渡环: 将线径进一步缩小
        'running'           -- 走线环: 竿稍部分小导环，传导信号
    )),
    guide_size REAL CHECK(guide_size > 0),          -- 导环尺寸/框号 (mm)
    weight REAL CHECK(weight >= 0),                 -- 单个重量 (g)，精确到0.01g
    total_length REAL CHECK(total_length > 0),      -- 总长度 (mm)

    -- ═══════════════════════════════════════════════════════════
    -- ⚡ 性能参数
    -- ═══════════════════════════════════════════════════════════
    anti_tangle_type TEXT CHECK(anti_tangle_type IN (
        'K-Frame',          -- K型防缠绕 (行业标准)
        '3D',               -- Kigan 3D防缠绕
        'MicroWave',        -- American Tackle微波导环
        '传统',
        '无'
    )),
    smoothness_score REAL CHECK(smoothness_score BETWEEN 1 AND 10),  -- 顺滑度评分 (1-10)
    friction_coefficient REAL CHECK(friction_coefficient >= 0),      -- 摩擦系数
    casting_boost_percent REAL CHECK(casting_boost_percent >= 0),    -- 抛投提升率 (%)
    sensitivity_score REAL CHECK(sensitivity_score BETWEEN 1 AND 10), -- 感度评分 (1-10)

    -- ═══════════════════════════════════════════════════════════
    -- 🌊 兼容性与耐用性
    -- ═══════════════════════════════════════════════════════════
    salt_water_resistance TEXT CHECK(salt_water_resistance IN ('优秀', '良好', '一般', '差')),
    compatible_line_types TEXT,             -- 兼容线组类型 (JSON): '["PE线", "尼龙线", "碳线"]'
    compatible_line_diameter TEXT,          -- 适配线径范围: '0.8-2.0号'
    max_line_lb REAL CHECK(max_line_lb > 0), -- 最大线磅数 (lb)
    wear_resistance TEXT CHECK(wear_resistance IN ('极高', '高', '中', '低')),

    -- ═══════════════════════════════════════════════════════════
    -- 🎯 适用场景
    -- ═══════════════════════════════════════════════════════════
    usage_scenario TEXT,                    -- 使用场景: 'Spinning Butt', 'Casting Tip', 'Heavy Duty'
    rod_type_compatibility TEXT,            -- 适配竿型 (JSON): '["纺车竿", "枪柄竿", "海钓竿"]'
    concept_compatibility TEXT,             -- 适配配导概念 (JSON): '["KR", "NGC"]'

    -- ═══════════════════════════════════════════════════════════
    -- 🔌 智能功能 (未来扩展)
    -- ═══════════════════════════════════════════════════════════
    smart_features BOOLEAN DEFAULT FALSE,
    sensor_types TEXT,                      -- 传感器类型 (JSON)

    -- ═══════════════════════════════════════════════════════════
    -- 🕐 元数据
    -- ═══════════════════════════════════════════════════════════
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (equipment_id) REFERENCES equipment(id) ON DELETE CASCADE,
    FOREIGN KEY (brand_id) REFERENCES guide_brands(id) ON DELETE SET NULL
);
```

### 2.2 SKU编码规则

遵循 Fuji 行业标准命名规则：

```
[支架材质代号]-[框型代号][磁环代号] [尺寸][高度标识]

示例: T-KLTG 20H
├── T      = Titanium (钛合金支架)
├── KL     = KL型框架 (单脚K型)
├── T      = Torzite (顶级磁环)
├── G      = Guide (导环)
├── 20     = 尺寸20号
└── H      = High frame (高脚)
```

**支架材质代号**:
| 代号 | 材质 | 说明 |
|------|------|------|
| **T** | Titanium | 钛合金 |
| **SS** | Stainless Steel | 不锈钢 |
| **C** | Carbon | 碳纤维 |
| **A** | Aluminum | 铝合金 |

**磁环材质代号**:
| 代号 | 材质 | 说明 |
|------|------|------|
| **T** | Torzite | 顶级 |
| **S** | SiC | 高端 |
| **N** | SiN | 重型 |
| **A** | Alconite | 中端 |
| **F** | FazLite | 入门进阶 |
| **O** | Aluminum Oxide | 入门 |

## 4. guide_train_concepts（配导系统表）

### 4.1 表结构

记录导环配置系统概念，这是产品开发的核心知识。

```sql
CREATE TABLE guide_train_concepts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    concept_code TEXT UNIQUE NOT NULL,      -- 概念代码: 'KR', 'NGC', 'SPIRAL'
    concept_name TEXT NOT NULL,             -- 概念名称
    concept_name_cn TEXT,                   -- 中文名称

    -- 技术原理
    principle TEXT,                         -- 工作原理
    key_feature TEXT,                       -- 核心特点
    stripper_guide_type TEXT,               -- 出发环类型: 'KL-H高脚小口径', '大口径'

    -- 适用场景
    suitable_rod_types TEXT,                -- 适用竿型 (JSON): '["微物BFS", "泛用路亚"]'
    suitable_line_types TEXT,               -- 适用线型 (JSON)

    -- 优势与劣势
    advantages TEXT,                        -- 优势 (JSON)
    disadvantages TEXT,                     -- 劣势 (JSON)

    -- 典型配置
    typical_guide_count_min INTEGER,        -- 典型导环数量下限
    typical_guide_count_max INTEGER,        -- 典型导环数量上限
    typical_configuration TEXT,             -- 典型配置示例 (JSON)

    introduction_year INTEGER,              -- 推出年份
    introduced_by TEXT,                     -- 推出品牌

    is_current_mainstream BOOLEAN DEFAULT FALSE, -- 是否当前主流
    description TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

### 4.2 配导系统概念详解

#### KR Concept (KR概念) - 当前主流

```
原理: 使用高脚、小口径的导环 (KL-H) 作为出发环
目的: 快速将渔轮出来的螺旋线束"扼杀"成直线
优势:
  ✅ 减少线在竿身上的拍打 (Line Slap)
  ✅ 极大提升感度
  ✅ 减轻竿体重量
适用: 微物(BFS)、泛用路亚竿
```

#### NGC (New Guide Concept) - 传统配置

```
原理: 使用口径较大的导环慢慢收缩线束
推出: Fuji 1995年
适用: 重型竿、长投竿
特点: 导环间距较大，口径逐渐递减
```

#### Spiral Wrap (螺旋导环) - 枪柄竿专用

```
做法: 导环从上方(0度)逐渐旋转到竿子下方(180度)
优势:
  ✅ 解决枪柄竿中鱼后线磨竿身的问题
  ✅ 重心更稳，不会扭竿
适用: 枪柄竿 (Baitcasting)
别名: 酸奶导环
```

## 5. rod_guide_kits（导环套装表）

### 5.1 表结构

预配置的导环套装，支持组合销售，v5.0 增加配导系统关联。

```sql
CREATE TABLE rod_guide_kits (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    kit_name TEXT NOT NULL,                 -- 套装名称
    kit_code TEXT UNIQUE NOT NULL,          -- 套装编码
    brand_id INTEGER,                       -- 关联guide_brands.id

    -- ═══════════════════════════════════════════════════════════

    -- 📦 套装规格
    -- ═══════════════════════════════════════════════════════════
    total_guides INTEGER NOT NULL CHECK(total_guides > 0), -- 导环总数

    tip_top_count INTEGER DEFAULT 1,        -- 顶环数量
    stripper_count INTEGER DEFAULT 1,       -- 出发环数量

    applicable_rod_length_min REAL,         -- 适用竿长下限 (m)
    applicable_rod_length_max REAL,         -- 适用竿长上限 (m)
    applicable_power TEXT,                  -- 适用硬度 (JSON): '["M", "MH", "H"]'

    -- ═══════════════════════════════════════════════════════════
    -- 💰 价格信息
    -- ═══════════════════════════════════════════════════════════
    kit_price REAL CHECK(kit_price >= 0),
    individual_total_price REAL CHECK(individual_total_price >= kit_price),
    savings_amount REAL CHECK(savings_amount >= 0),
    savings_percentage REAL CHECK(savings_percentage BETWEEN 0 AND 100),

    -- ═══════════════════════════════════════════════════════════
    -- 📝 描述与状态
    -- ═══════════════════════════════════════════════════════════
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (brand_id) REFERENCES guide_brands(id) ON DELETE SET NULL,
    FOREIGN KEY (concept_id) REFERENCES guide_train_concepts(id) ON DELETE SET NULL
);
```

## 6. rod_guide_kit_items（套装明细表）

### 6.1 表结构

导环套装的具体组成明细。

```sql
CREATE TABLE rod_guide_kit_items (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    kit_id INTEGER NOT NULL,                -- 关联rod_guide_kits.id
    guide_equipment_id TEXT NOT NULL,       -- 关联导环equipment.id

    -- ═══════════════════════════════════════════════════════════
    -- 📍 位置信息
    -- ═══════════════════════════════════════════════════════════
    position_index INTEGER CHECK(position_index > 0), -- 安装位置索引 (1=竿稍)
    position_name TEXT CHECK(position_name IN (
        'TIP_TOP',          -- 顶环位置
        'STRIPPER',         -- 出发环位置
        'CHOKE_1',          -- 第1过渡环
        'CHOKE_2',          -- 第2过渡环
        'RUNNING_1',        -- 第1走线环
        'RUNNING_2',        -- 第2走线环
        'RUNNING_3',        -- 第3走线环
        'RUNNING_4',        -- 第4走线环
        'RUNNING_5',        -- 第5走线环
        'RUNNING_6',        -- 第6走线环
        'BELLY'             -- 腹部位置 (高负荷)
    )),
    distance_from_tip_mm INTEGER CHECK(distance_from_tip_mm >= 0), -- 距竿稍距离 (mm)

    -- ═══════════════════════════════════════════════════════════
    -- 🔢 数量与排序
    -- ═══════════════════════════════════════════════════════════
    quantity INTEGER DEFAULT 1 CHECK(quantity > 0),
    sort_order INTEGER DEFAULT 0,

    -- ═══════════════════════════════════════════════════════════
    -- 🔄 Spiral Wrap 支持
    -- ═══════════════════════════════════════════════════════════
    rotation_angle INTEGER DEFAULT 0 CHECK(rotation_angle BETWEEN 0 AND 180), -- 螺旋角度 (0-180度)

    -- ═══════════════════════════════════════════════════════════
    -- 🕐 元数据
    -- ═══════════════════════════════════════════════════════════

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (kit_id) REFERENCES rod_guide_kits(id) ON DELETE CASCADE,
    FOREIGN KEY (guide_equipment_id) REFERENCES equipment(id) ON DELETE CASCADE,
    UNIQUE (kit_id, position_index)
```

## 7. 专业知识详解

### 7.1 磁环材质对照表 (Ring Materials)

磁环直接接触钓线，决定了**散热性（防断线）**、**硬度（防磨损）**和**顺滑度（抛投距离）**。

| 材质 | 代号 | 硬度 (Hv) | 导热性 | 产品定位 | 典型价格 |
|------|------|-----------|--------|----------|----------|
| **Torzite** | T | 顶级 | 极高 | 顶级旗舰 ($300+ USD竿) | 最贵 |
| **SiC (碳化硅)** | S | ~2200 | 高 | 高端标配 | 高 |
| **SiN (氮化硅)** | N | 高 | 中 | 重型海钓/雷强 | 高 |
| **Alconite** | A | ~1500 | 中 | 中端性价比王 | 中 |
| **FazLite** | F | 中 | 中 | 入门进阶 | 中低 |
| **Zirconia** | - | 中高 | 中 | 非Fuji系主流 | 中 |
| **Aluminum Oxide** | O | 低 | 低 | 低端入门 | 低 |

**产品开发关键点**:
- **成本陷阱**: Torzite环成本极高，通常只用于$300 USD以上的竿子
- **轻量化优势**: Torzite比SiC薄约40%，同框号下内径更大，可用更小号导环达到同样内径

### 7.2 支架材质对照表 (Frame Materials)

| 材质 | 特性 | 产品定位 | 典型搭配 |
|------|------|----------|----------|
| **钛合金** | 极轻，不生锈，记忆弹性 | 顶级 | Torzite/SiC |
| **碳纤维** | 最轻，高刚性，感度最强 | 顶级 (Daiwa AGS专利) | - |
| **不锈钢 316L** | 高防锈等级 | 主流高端 | SiC/Alconite |
| **不锈钢 304** | 标准防锈 | 主流中端 | Alconite/FazLite |
| **铝合金** | 轻量，经济 | 入门 | FazLite/O环 |

### 7.3 框型几何分类 (Frame Types)

#### K系列 - 行业标准

| 型号 | 脚型 | 特点 | 使用位置 |
|------|------|------|----------|
| **KW** | 双脚 | 强力型 | 枪柄竿、大物竿 |
| **KL** | 单脚 | 高脚，可替代双脚 | 通用 |
| **KL-H** | 单脚 | 高脚小口径，KR概念专用 | 出发环位置 |
| **KT** | 单脚 | 极短 | 竿稍走线环 |
| **KB** | 单脚 | 加宽脚，承受高负荷 | 腹部位置 (Belly) |

#### 其他系列

| 型号 | 特点 | 使用场景 |
|------|------|----------|
| **Y系列** | 传统高支架 | 旧式/低端纺车竿 |
| **MN系列** | 重型双脚包覆式 | GT竿、铁板竿 |
| **L系列** | 极简微物 | 极轻量化竿 |
| **RV系列** | 钛合金反向双脚 | 顶级出发环，超远投 |

### 7.4 导环位置功能说明

| 位置 | 英文名 | 功能 | 关键特点 |
|------|--------|------|----------|
| **顶环** | Tip Top | 竿稍最前端 | 必须有防缠绕设计 |
| **出发环** | Stripper/Butt Guide | 离轮座最近 | 收束螺旋线，最大口径 |
| **过渡环** | Choke/Reduction | 中间段 | 将线径进一步缩小 |
| **走线环** | Running Guide | 竿稍部分 | 传导信号，分散拉力 |

### 9.1 按材质和产品定位查询导环

```python
def search_guides(
    self,
    ring_material: Optional[str] = None,
    frame_material: Optional[str] = None,
    product_tier: Optional[str] = None,
    guide_type: Optional[str] = None,
    brand_code: Optional[str] = None,
    size_min: Optional[float] = None,
    size_max: Optional[float] = None,
    limit: int = 20
):
    """查询导环产品 (v5.0)"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT
            e.id, e.name, e.brand, e.price_min, e.price_max,
            gs.*,
            gb.brand_name as guide_brand_name,
            gb.market_position
        FROM equipment e
        JOIN guide_specs gs ON e.id = gs.equipment_id
        LEFT JOIN guide_brands gb ON gs.brand_id = gb.id
        WHERE e.category = '导环'
    '''
    params = []

    if ring_material:
        query += ' AND gs.ring_material = ?'
        params.append(ring_material)

    if frame_material:
        query += ' AND gs.frame_material = ?'
        params.append(frame_material)

    if product_tier:
        query += ' AND gs.product_tier = ?'
        params.append(product_tier)

    if guide_type:
        query += ' AND gs.guide_type = ?'
        params.append(guide_type)

    if brand_code:
        query += ' AND gb.brand_code = ?'
        params.append(brand_code)

    if size_min is not None and size_max is not None:
        query += ' AND gs.guide_size BETWEEN ? AND ?'
        params.extend([size_min, size_max])

    query += ' ORDER BY gs.product_tier, e.price_min LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)
    return [dict(row) for row in cursor.fetchall()]
```

### 9.2 查询配导套装

```python
def get_guide_kits_by_concept(
    self,
    concept_type: str,
    rod_length: Optional[float] = None,
    power: Optional[str] = None
):
    """按配导概念查询套装"""
    conn = self._get_connection()
    cursor = conn.cursor()

    query = '''
        SELECT
            rgk.*,
            gb.brand_name,
            gtc.concept_name_cn,
            gtc.principle as concept_principle
        FROM rod_guide_kits rgk
        LEFT JOIN guide_brands gb ON rgk.brand_id = gb.id
        LEFT JOIN guide_train_concepts gtc ON rgk.concept_id = gtc.id
        WHERE rgk.is_active = 1
          AND rgk.concept_type = ?
    '''
    params = [concept_type]

    if rod_length:
        query += '''
            AND rgk.applicable_rod_length_min <= ?
            AND rgk.applicable_rod_length_max >= ?
        '''
        params.extend([rod_length, rod_length])

    if power:
        query += " AND rgk.applicable_power LIKE ?"
        params.append(f'%"{power}"%')

    query += ' ORDER BY rgk.total_weight ASC'

    cursor.execute(query, params)
    return [dict(row) for row in cursor.fetchall()]
```

### 9.3 获取套装完整配置

```python
def get_kit_full_configuration(self, kit_id: int):
    """获取导环套装完整配置 (含每个位置详情)"""
    conn = self._get_connection()
    cursor = conn.cursor()

    # 获取套装基本信息
    cursor.execute('''
        SELECT
            rgk.*,
            gb.brand_name,
            gtc.concept_name_cn,
            gtc.principle
        FROM rod_guide_kits rgk
        LEFT JOIN guide_brands gb ON rgk.brand_id = gb.id
        LEFT JOIN guide_train_concepts gtc ON rgk.concept_id = gtc.id
        WHERE rgk.id = ?
    ''', (kit_id,))
    kit_info = dict(cursor.fetchone())

    # 获取各位置导环配置
    cursor.execute('''
        SELECT
            rgki.position_index,
            rgki.position_name,
            rgki.quantity,
            rgki.distance_from_tip_mm,
            rgki.rotation_angle,
            e.name as guide_name,
            e.brand as guide_brand,
            gs.sku_code,
            gs.guide_size,
            gs.ring_material,
            gs.frame_material,
            gs.frame_type,
            gs.weight
        FROM rod_guide_kit_items rgki
        JOIN equipment e ON rgki.guide_equipment_id = e.id
        LEFT JOIN guide_specs gs ON rgki.guide_equipment_id = gs.equipment_id
        WHERE rgki.kit_id = ?
        ORDER BY rgki.position_index
    ''', (kit_id,))
    guides = [dict(row) for row in cursor.fetchall()]

    return {
        'kit_info': kit_info,
        'guides': guides,
        'total_weight': sum(g['weight'] * g['quantity'] for g in guides if g['weight'])
    }
```
