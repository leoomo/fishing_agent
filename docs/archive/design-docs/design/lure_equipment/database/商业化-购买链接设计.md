## 🎯 业务需求

### 核心需求
- **链接数量**：每个装备最多展示3个购买链接
- **价格更新**：定时爬取（每天/每周自动更新）
- **排序策略**：按平台优先级+价格排序（不考虑佣金）
- **合作平台**：京东联盟、淘宝联盟（主推）、拼多多多多进宝、抖音

### 使用场景

**用户查询**：
```
"请帮我推荐重量小于100克、超快调国产鱼竿"
```

**系统返回**：
```
【光威竹山先调鱼竿】92克 超快调
购买链接：
1. [京东自营] ¥399 - 11.3%折扣
2. [拼多多] ¥380 - 百亿补贴
3. [淘宝官旗] ¥420
```

## 🗄️ 表结构设计

### equipment_purchase_links（装备购买链接表）

```sql
CREATE TABLE equipment_purchase_links (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    equipment_id TEXT NOT NULL,              -- 关联equipment.id

    -- 平台信息
    platform TEXT NOT NULL CHECK(platform IN ('京东', '淘宝', '天猫', '拼多多', '抖音', '其他')),
    platform_priority INTEGER DEFAULT 0,    -- 平台优先级（京东/淘宝=1，拼多多=2，抖音=3）
    merchant_name TEXT,                      -- 商家名称

    -- 购买链接
    purchase_url TEXT NOT NULL,              -- 推广链接（含联盟参数）
    short_url TEXT,                          -- 短链接（便于分享）
    qr_code_url TEXT,                        -- 二维码图片URL

    -- 价格信息
    current_price REAL CHECK(current_price >= 0),    -- 当前价格
    original_price REAL,                     -- 原价
    discount_rate REAL CHECK(discount_rate BETWEEN 0 AND 100),  -- 折扣率(%)

    -- 定时爬取相关
    price_last_updated TIMESTAMP,            -- 价格最后更新时间
    price_update_frequency INTEGER DEFAULT 24,  -- 更新频率（小时，默认24小时）
    auto_update_enabled BOOLEAN DEFAULT TRUE,   -- 是否启用自动更新

    -- 展示与状态
    is_active BOOLEAN DEFAULT TRUE,          -- 是否有效
    valid_from TIMESTAMP,                    -- 生效时间
    valid_until TIMESTAMP,                   -- 失效时间（促销结束）
    badge_text TEXT,                         -- 展示标签（"自营"、"官旗"、"百亿补贴"）
    promotion_type TEXT,                     -- 促销类型（双11、618等）

    -- 统计数据（不影响排序）
    click_count INTEGER DEFAULT 0,           -- 点击次数
    purchase_count INTEGER DEFAULT 0,        -- 购买次数（需要回调）
    last_clicked_at TIMESTAMP,               -- 最后点击时间

    -- 元数据
    notes TEXT,                              -- 备注
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (equipment_id) REFERENCES equipment(id) ON DELETE CASCADE
);

-- 索引设计
CREATE INDEX idx_purchase_equipment ON equipment_purchase_links(equipment_id);
CREATE INDEX idx_purchase_platform ON equipment_purchase_links(platform);
CREATE INDEX idx_purchase_active_price ON equipment_purchase_links(is_active, current_price);
CREATE INDEX idx_purchase_update_time ON equipment_purchase_links(price_last_updated);
CREATE INDEX idx_purchase_priority ON equipment_purchase_links(platform_priority, current_price);
```

### 插入时自动设置优先级

```python
def add_purchase_link(self, link_data: dict) -> int:
    """添加购买链接（自动设置平台优先级）"""
    conn = self._get_connection()
    cursor = conn.cursor()

    # 自动设置平台优先级
    link_data['platform_priority'] = PLATFORM_PRIORITY.get(
        link_data['platform'], 99
    )

    try:
        cursor.execute('''
            INSERT INTO equipment_purchase_links (
                equipment_id, platform, platform_priority, purchase_url,
                current_price, original_price, badge_text, auto_update_enabled
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        ''', (
            link_data['equipment_id'],
            link_data['platform'],
            link_data['platform_priority'],
            link_data['purchase_url'],
            link_data.get('current_price'),
            link_data.get('original_price'),
            link_data.get('badge_text'),
            link_data.get('auto_update_enabled', True)
        ))
        conn.commit()
        return cursor.lastrowid
    except Exception as e:
        conn.rollback()
        raise
```

## 📊 核心查询方法

### 1. 获取购买链接（限制3个）

```python
def get_purchase_links(self, equipment_id: str, limit: int = 3) -> List[Dict]:
    """
    获取装备的购买链接（最多3个）

    排序规则：
    1. 平台优先级（京东/淘宝 > 拼多多 > 抖音）
    2. 价格升序（最低价优先）

    Args:
        equipment_id: 装备ID
        limit: 返回链接数量（默认3个）

    Returns:
        购买链接列表
    """
    conn = self._get_connection()
    cursor = conn.cursor()

    cursor.execute('''
        SELECT
            id,
            platform,
            purchase_url,
            short_url,
            current_price,
            original_price,
            discount_rate,
            badge_text,
            price_last_updated,
            promotion_type,
            valid_until
        FROM equipment_purchase_links
        WHERE equipment_id = ?
          AND is_active = 1
          AND (valid_until IS NULL OR valid_until > CURRENT_TIMESTAMP)
        ORDER BY
            platform_priority ASC,    -- 京东/淘宝优先
            current_price ASC         -- 价格最低优先
        LIMIT ?
    ''', (equipment_id, limit))

    links = []
    for row in cursor.fetchall():
        link = dict(row)
        # 计算折扣率（如果未预存）
        if link['original_price'] and not link['discount_rate']:
            link['discount_rate'] = round(
                (1 - link['current_price'] / link['original_price']) * 100, 1
            )
        links.append(link)

    return links
```

### 2. 批量获取多个装备的购买链接

```python
def get_batch_purchase_links(self, equipment_ids: List[str], limit_per_item: int = 3) -> Dict:
    """
    批量获取多个装备的购买链接

    Args:
        equipment_ids: 装备ID列表
        limit_per_item: 每个装备返回链接数

    Returns:
        {equipment_id: [links]}
    """
    result = {}
    for equipment_id in equipment_ids:
        result[equipment_id] = self.get_purchase_links(equipment_id, limit_per_item)

    return result
```

### 3. 推荐鱼竿并返回购买链接（完整示例）

```python
def recommend_rods_with_purchase_links(
    self,
    fishing_type: Optional[str] = None,
    power: Optional[str] = None,
    weight_max: Optional[float] = None,
    action: Optional[str] = None,
    tier: Optional[str] = None,
    price_max: Optional[float] = None,
    limit: int = 10
) -> List[Dict]:
    """
    推荐鱼竿并返回购买链接

    使用示例：
    results = db.recommend_rods_with_purchase_links(
        weight_max=100,    # 重量小于100克
        action='XF',       # 超快调
        tier='国产'        # 国产品牌
    )
    """
    conn = self._get_connection()
    cursor = conn.cursor()

    # 构建查询条件
    query = '''
        SELECT
            e.id, e.name, e.brand, e.price_min, e.price_max,
            r.length, r.power, r.action, r.weight,
            r.fishing_type, r.water_type,
            r.carbon_grade, r.material_description,
            r.lure_weight_raw,
            b.reputation_score, b.tier, b.country
        FROM equipment e
        JOIN rod_specs r ON e.id = r.equipment_id
        JOIN brands b ON e.brand = b.brand_name
        WHERE e.category = '鱼竿'
    '''
    params = []

    if fishing_type:
        query += ' AND r.fishing_type = ?'
        params.append(fishing_type)

    if power:
        query += ' AND r.power = ?'
        params.append(power)

    if weight_max:
        query += ' AND r.weight < ?'
        params.append(weight_max)

    if action:
        query += ' AND r.action = ?'
        params.append(action)

    if tier:
        query += ' AND b.tier = ?'
        params.append(tier)

    if price_max:
        query += ' AND e.price_min <= ?'
        params.append(price_max)

    query += ' ORDER BY b.reputation_score DESC, e.confidence DESC LIMIT ?'
    params.append(limit)

    cursor.execute(query, params)
    rods = [dict(row) for row in cursor.fetchall()]

    # 为每个鱼竿获取购买链接（最多3个）
    for rod in rods:
        rod['purchase_links'] = self.get_purchase_links(rod['id'], limit=3)

    return rods
```

## 🔄 定时爬取价格更新

### 方案1：简单定时任务（推荐MVP阶段）

```python
import schedule
import time
import logging

logger = logging.getLogger(__name__)


def update_all_prices(db):
    """更新所有启用自动更新的链接价格"""
    conn = db._get_connection()
    cursor = conn.cursor()

    # 获取需要更新的链接
    cursor.execute('''
        SELECT id, equipment_id, platform, purchase_url, price_update_frequency
        FROM equipment_purchase_links
        WHERE auto_update_enabled = 1
          AND is_active = 1
          AND (
              price_last_updated IS NULL
              OR (julianday('now') - julianday(price_last_updated)) * 24 >= price_update_frequency
          )
    ''')

    links_to_update = cursor.fetchall()
    logger.info(f"准备更新 {len(links_to_update)} 个购买链接的价格")

    success_count = 0
    fail_count = 0

    for link in links_to_update:
        try:
            # 调用爬虫获取最新价格
            new_price_data = crawl_price(link['platform'], link['purchase_url'])

            # 更新价格
            cursor.execute('''
                UPDATE equipment_purchase_links
                SET current_price = ?,
                    original_price = ?,
                    discount_rate = ?,
                    price_last_updated = CURRENT_TIMESTAMP,
                    updated_at = CURRENT_TIMESTAMP
                WHERE id = ?
            ''', (
                new_price_data['current_price'],
                new_price_data.get('original_price'),
                new_price_data.get('discount_rate'),
                link['id']
            ))

            success_count += 1
            logger.debug(f"更新成功: {link['id']} - 新价格: {new_price_data['current_price']}")

        except Exception as e:
            fail_count += 1
            logger.error(f"更新失败: {link['id']}, 错误: {e}")

    conn.commit()
    logger.info(f"价格更新完成 - 成功: {success_count}, 失败: {fail_count}")
```

```python
from celery.schedules import crontab

app = Celery('lure_equipment_tasks', broker='redis://localhost:6379/0')
```
