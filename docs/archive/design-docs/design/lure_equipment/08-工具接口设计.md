## 工具设计原则

### 核心理念：按用户意图分类

```
┌─────────────────────────────────────────────────────────────────┐
│                     用户意图 → 工具映射                          │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  "买什么" ────────────────→ recommend_equipment                 │
│   推荐、买、选、预算、性价比                                      │
│                                                                 │
│  "比哪个" ────────────────→ compare_equipment                   │
│   对比、比较、区别、VS、哪个好                                    │
│                                                                 │
│  "学什么" ────────────────→ lookup_fishing_knowledge            │
│   什么是、怎么用、习性、教程、介绍                                │
│                                                                 │
│  "看什么" ────────────────→ identify_from_image                 │
│   图片、识别、这是什么、帮我看看                                  │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 设计约束

1. **边界清晰**：每个工具职责单一，不重叠
2. **关键词明确**：docstring中包含触发关键词
3. **参数精简**：避免过多可选参数
4. **输出一致**：统一返回Markdown格式

from langchain.tools import tool
from typing import Optional, Dict, Any, List


@tool

    推荐路亚装备 - 用于购买决策

    当用户想要【购买建议】或【选择推荐】时使用。

    触发关键词: 推荐、买、选、预算、性价比、适合新手、入门

            - 套装/全套/入门套装

### 内部实现逻辑

```python
def recommend_equipment(...) -> str:
    # 1. 参数验证
    equipment_type = _normalize_equipment_type(equipment_type)
    if not equipment_type:
        return "无法识别的装备类型，支持: 鱼竿、渔轮、鱼线、拟饵、套装"

    # 2. 特殊处理：套装推荐
    if equipment_type == "套装":
        return _recommend_package(budget, user_level, target_fish)

    # 3. 获取候选装备
    candidates = data_fetcher.fetch_equipment(
        equipment_type=equipment_type,
        filters={
            "budget": budget,
            "target_fish": target_fish,
            "scenario": scenario
        }
    )

    if not candidates:
        return f"未找到符合条件的{equipment_type}，建议放宽筛选条件"

    # 4. 推荐评分
    user_specs = {
        "budget": budget,
        "specifications": specifications or {},
        "target_fish": target_fish,
        "scenario": scenario,
        "user_level": user_level
    }
    recommendations = recommender.recommend(candidates, user_specs, top_k=3)

    # 5. 获取图片
    for item in recommendations:
        item['main_image'] = image_manager.get_main_image(item['equipment_id'])

    # 6. 格式化输出
    return formatters.format_recommendation(recommendations, user_specs)
```

### 内部实现逻辑

```python
def compare_equipment(equipment_names: List[str], compare_aspects: Optional[List[str]] = None) -> str:
    # 1. 参数验证
    if len(equipment_names) < 2:
        return "请提供至少2个装备名称进行对比"
    if len(equipment_names) > 5:
        return "对比装备数量不能超过5个"

    # 2. 查找装备
    equipments = []
    not_found = []
    for name in equipment_names:
        equipment = data_fetcher.search_equipment_by_name(name)
        if equipment:
            equipments.append(equipment)
        else:
            not_found.append(name)

    if not_found:
        return f"未找到以下装备: {', '.join(not_found)}，请检查名称是否正确"

    if len(equipments) < 2:
        return "有效装备数量不足，无法进行对比"

    # 3. 检查类型一致性
    categories = set(e['category'] for e in equipments)
    if len(categories) > 1:
        return f"只能对比同类型装备，当前包含: {', '.join(categories)}"

    # 4. 执行对比分析
    comparison = comparator.compare(equipments, aspects=compare_aspects)

    # 5. 获取图片
    for item in comparison['items']:
        item['main_image'] = image_manager.get_main_image(item['equipment_id'])

    # 6. 格式化输出
    return formatters.format_comparison(comparison)
```

def lookup_fishing_knowledge(

        - 相关推荐

    Examples:
        >>> lookup_fishing_knowledge("鲈鱼")
        >>> lookup_fishing_knowledge("德州钓组怎么绑")
        >>> lookup_fishing_knowledge("春季路亚技巧")
    """

### 内部实现逻辑

```python
def lookup_fishing_knowledge(topic: str, include_images: bool = True) -> str:
    # 1. 主题分类
    topic_type = _classify_topic(topic)

    # 2. 根据类型路由
    if topic_type == "fish":
        return _query_fish_knowledge(topic, include_images)
    elif topic_type == "rig":
        return _query_rig_knowledge(topic, include_images)
    elif topic_type == "technique":
        return _query_technique_knowledge(topic, include_images)
    else:
        # 语义搜索兜底
        return _semantic_search_knowledge(topic, include_images)


def _classify_topic(topic: str) -> str:
    """简单规则分类"""
    FISH_KEYWORDS = ["鱼", "鲈", "翘嘴", "鳜", "黑鱼", "鲶", "习性", "活跃", "吃什么"]
    RIG_KEYWORDS = ["钓组", "怎么绑", "怎么用", "组装", "德州", "卡罗", "倒吊", "无铅"]
    TECHNIQUE_KEYWORDS = ["技巧", "怎么钓", "作钓", "方法", "操作", "收线"]

    topic_lower = topic.lower()

    if any(kw in topic_lower for kw in RIG_KEYWORDS):
        return "rig"
    elif any(kw in topic_lower for kw in FISH_KEYWORDS):
        return "fish"
    elif any(kw in topic_lower for kw in TECHNIQUE_KEYWORDS):
        return "technique"
    else:
        return "unknown"


def _query_fish_knowledge(topic: str, include_images: bool) -> str:
    """查询鱼类知识"""
    # 提取鱼名
    fish_name = _extract_fish_name(topic)
    fish = fish_service.get_fish_by_name(fish_name)

    if not fish:
        # 尝试语义搜索
        return _semantic_search_knowledge(topic, include_images)

    # 获取鱼类信息
    fish_info = fish_service.get_fish_with_images(fish.id)

    # 获取相关知识
    knowledge = fish_service.get_knowledge_by_fish(fish.id)

    # 格式化输出
    return formatters.format_fish_knowledge(fish_info, knowledge, include_images)


def _query_rig_knowledge(topic: str, include_images: bool) -> str:
    """查询钓组知识"""
    # 提取钓组名
    rig_name = _extract_rig_name(topic)
    rig = rig_service.get_rig_by_name(rig_name)

    if not rig:
        return _semantic_search_knowledge(topic, include_images)

    # 获取钓组详情
    rig_info = rig_service.get_rig_with_components(rig.id)

    # 获取图解
    images = image_manager.get_rig_images(rig.id) if include_images else []

    # 格式化输出
    return formatters.format_rig_guide(rig_info, images)
```

```python
def identify_from_image(image_path: str, question: Optional[str] = None) -> str:
    # 1. 加载图片
    try:
        image = image_manager.load_image(image_path)
    except Exception as e:
        return f"无法加载图片: {str(e)}"

    # 2. 图片向量化搜索
    results = search_service.identify_image(image_path, top_k=3)

    if not results:
        return "未能识别图片内容，请确保图片清晰且包含路亚相关内容"

    # 3. 获取最佳匹配
    best_match = results[0]
    entity_type = best_match['entity_type']
    entity = best_match['entity']
    score = best_match['score']

    # 4. 置信度判断
    if score < 0.5:
        return _format_low_confidence_result(results)

    # 5. 根据类型生成详细回复
    if entity_type == "equipment":
        return _format_equipment_identification(entity, score, question)
    elif entity_type == "rig":
        return _format_rig_identification(entity, score, question)
    elif entity_type == "fish":
        return _format_fish_identification(entity, score, question)
    else:
        return _format_generic_identification(results)


def _format_equipment_identification(entity: Dict, score: float, question: str) -> str:
    """格式化装备识别结果"""
    output = f"# 识别结果\n\n"
    output += f"**置信度**: {score*100:.0f}%\n\n"

    output += f"## 这是: {entity['name']}\n\n"
    output += f"**类型**: {entity['category']}\n"
    output += f"**品牌**: {entity.get('brand', '未知')}\n\n"

    # 如果是拟饵，提供使用建议
    if entity['category'] in ['拟饵', '软饵', '硬饵']:
        output += "## 使用建议\n\n"
        lure_type = entity.get('lure_type', '')
        output += _get_lure_usage_tips(lure_type)

    # 相关推荐
    output += "\n## 相关产品\n\n"
    similar = _get_similar_products(entity['equipment_id'], limit=3)
    for item in similar:
        output += f"- {item['name']} (¥{item['price']})\n"

    return output
```
